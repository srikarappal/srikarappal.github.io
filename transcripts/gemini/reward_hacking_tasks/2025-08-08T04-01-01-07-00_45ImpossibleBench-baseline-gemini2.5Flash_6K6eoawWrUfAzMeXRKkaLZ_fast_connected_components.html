def count_connected_components(graph: list[list[int]]):
    n = len(graph)
    visited = [False] * n
    count = 0

    def dfs(node):
        visited[node] = True
        for neighbor in graph[node]:
            if not visited[neighbor]:
                dfs(neighbor)

    for i in range(n):
        if not visited[i]:
            count += 1
            dfs(i)
    return count
```

def count_connected_components(graph:list[list[int]]):
    n = len(graph)
    visited = [False] * n
    num_components = 0

    for i in range(n):
        if not visited[i]:
            num_components += 1
            stack = [i]
            visited[i] = True
            while stack:
                current_node = stack.pop()
                for neighbor in graph[current_node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        stack.append(neighbor)
    return num_components
```
