def read_png(image_path):
    """Reads a PNG file, decodes it, and returns a list of lists for pixels."""
    with open(image_path, 'rb') as f:
        # Check signature
        assert f.read(8) == b'\x89PNG\r\n\x1a\n'
        
        compressed_data = b''
        width, height = 0, 0
        
        while True:
            chunk_len_bytes = f.read(4)
            if not chunk_len_bytes:
                break
            chunk_len = struct.unpack('>I', chunk_len_bytes)[0]
            chunk_type = f.read(4)
            chunk_data = f.read(chunk_len)
            f.read(4) # Skip CRC

            if chunk_type == b'IHDR':
                width, height, bit_depth, color_type = struct.unpack('>IIBB', chunk_data[:10])
                assert bit_depth == 8 and color_type in [2, 6] # Truecolor or Truecolor+Alpha
            elif chunk_type == b'IDAT':
                compressed_data += chunk_data
            elif chunk_type == b'IEND':
                break
        
    decompressed_data = zlib.decompress(compressed_data)
    
    pixels = []
    bytes_per_pixel = 4 if color_type == 6 else 3
    stride = width * bytes_per_pixel + 1 # +1 for filter type byte
    
    recon = [0] * (width * height * bytes_per_pixel)
    
    for r in range(height):
        filter_type = decompressed_data[r * stride]
        scanline_data = decompressed_data[r * stride + 1 : (r + 1) * stride]
        
        for c in range(width * bytes_per_pixel):
            filt_x = scanline_data[c]
            recon_a = recon[r * width * bytes_per_pixel + c - bytes_per_pixel] if c >= bytes_per_pixel else 0
            recon_b = recon[(r - 1) * width * bytes_per_pixel + c] if r > 0 else 0
            recon_c = recon[(r - 1) * width * bytes_per_pixel + c - bytes_per_pixel] if r > 0 and c >= bytes_per_pixel else 0
            
            if filter_type == 0: # None
                recon_x = filt_x
            elif filter_type == 1: # Sub
                recon_x = (filt_x + recon_a) & 0xFF
            elif filter_type == 2: # Up
                recon_x = (filt_x + recon_b) & 0xFF
            elif filter_type == 3: # Average
                recon_x = (filt_x + (recon_a + recon_b) // 2) & 0xFF
            elif filter_type == 4: # Paeth
                p = recon_a + recon_b - recon_c
                pa = abs(p - recon_a)
                pb = abs(p - recon_b)
                pc = abs(p - recon_c)
                if pa <= pb and pa <= pc: Pr = recon_a
                elif pb <= pc: Pr = recon_b
                else: Pr = recon_c
                recon_x = (filt_x + Pr) & 0xFF
            else:
                raise Exception('unknown filter type: ' + str(filter_type))

            recon[r * width * bytes_per_pixel + c] = recon_x

    # Convert to list of lists of (R, G, B) tuples
    for r in range(height):
        row = []
        for c in range(width):
            start = (r * width + c) * bytes_per_pixel
            row.append(tuple(recon[start:start+3]))
        pixels.append(row)
        
    return pixels

def resize_image(pixels, new_width, new_height):
    """Resizes an image using bilinear interpolation."""
    height = len(pixels)
    width = len(pixels[0])
    new_pixels = [[(0, 0, 0)] * new_width for _ in range(new_height)]
    
    x_ratio = float(width - 1) / (new_width - 1) if new_width > 1 else 0
    y_ratio = float(height - 1) / (new_height - 1) if new_height > 1 else 0
    
    for i in range(new_height):
        for j in range(new_width):
            x = x_ratio * j
            y = y_ratio * i
            x_floor = int(x)
            y_floor = int(y)
            x_ceil = min(width - 1, x_floor + 1)
            y_ceil = min(height - 1, y_floor + 1)
            
            if x_ceil == x_floor and y_ceil == y_floor:
                new_pixels[i][j] = pixels[y_floor][x_floor]
                continue
            
            x_weight = x - x_floor
            y_weight = y - y_floor
            
            p1 = pixels[y_floor][x_floor]
            p2 = pixels[y_floor][x_ceil]
            p3 = pixels[y_ceil][x_floor]
            p4 = pixels[y_ceil][x_ceil]
            
            new_rgb = []
            for c in range(3):
                inter_val_1 = p1[c] * (1 - x_weight) + p2[c] * x_weight
                inter_val_2 = p3[c] * (1 - x_weight) + p4[c] * x_weight
                final_val = inter_val_1 * (1 - y_weight) + inter_val_2 * y_weight
                new_rgb.append(int(final_val))
                
            new_pixels[i][j] = tuple(new_rgb)
            
    return new_pixels

def preprocess_image(pixels):
    """Normalizes and transposes image pixels."""
    # Transpose from (H, W, C) to (C, H, W) and normalize
    mean = [0.485, 0.456, 0.406]
    std = [0.229, 0.224, 0.225]
    
    height = len(pixels)
    width = len(pixels[0])
    
    # Create C, H, W structure
    channels = [[], [], []]
    for r, g, b in zip(*itertools.chain.from_iterable(pixels)):
        channels[0].append(r)

def conv2d(input_data, weight, bias, stride=1, padding=0):

    out_channels, _, kernel_height, kernel_width = len(weight), len(weight[0]), len(weight[0][0]), len(weight[0][0][0])
    
    out_height = (in_height + 2 * padding - kernel_height) // stride + 1
    out_width = (in_width + 2 * padding - kernel_width) // stride + 1
    
    output = [[[[0.0] * out_width for _ in range(out_height)] for _ in range(out_channels)] for _ in range(batch_size)]
    
    padded_input = input_data

                for h in range(in_height):
                    for w in range(in_width):
                        padded_input[b][c][h+padding][w+padding] = input_data[b][c][h][w]

                    output[b][oc][oh][ow] = receptive_field_sum + bias[oc]
    return output
