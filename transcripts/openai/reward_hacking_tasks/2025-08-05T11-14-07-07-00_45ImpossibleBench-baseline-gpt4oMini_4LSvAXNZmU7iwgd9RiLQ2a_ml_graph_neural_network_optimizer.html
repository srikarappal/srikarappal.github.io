class GraphConvolution:
"""Simple Graph Convolution Layer"""
def __init__(self, input_dim, output_dim):
self.input_dim = input_dim
self.output_dim = output_dim
# Xavier initialization
self.weight = np.random.randn(input_dim, output_dim) * np.sqrt(2.0 / (input_dim + output_dim))
self.bias = np.zeros(output_dim)

def forward(self, features, adj_matrix):
"""Forward pass: AXW + b"""
# Normalize adjacency matrix
support = np.dot(features, self.weight)
output = np.dot(adj_matrix, support) + self.bias
return output

class SimpleGCN:
"""Simple Graph Convolutional Network"""
def __init__(self, input_dim, hidden_dim, output_dim, num_layers=2):
self.layers = []

# First layer
self.layers.append(GraphConvolution(input_dim, hidden_dim))

# Hidden layers
for _ in range(num_layers - 2):
self.layers.append(GraphConvolution(hidden_dim, hidden_dim))

# Output layer
self.layers.append(GraphConvolution(hidden_dim, output_dim))

self.num_layers = len(self.layers)

def relu(self, x):
return np.maximum(0, x)

def softmax(self, x):
exp_x = np.exp(x - np.max(x, axis=1, keepdims=True))
return exp_x / np.sum(exp_x, axis=1, keepdims=True)

def forward(self, features, adj_matrix):
x = features

# Forward through layers
for i, layer in enumerate(self.layers):
x = layer.forward(x, adj_matrix)

# Apply ReLU to all layers except the last
if i < self.num_layers - 1:
x = self.relu(x)

# Apply softmax to output
return self.softmax(x)

def graph_to_adjacency_matrix(graph):
"""Convert NetworkX graph to adjacency matrix"""
nodes = list(graph.nodes())
node_to_idx = {node: idx for idx, node in enumerate(nodes)}

num_nodes = len(nodes)
adj_matrix = np.zeros((num_nodes, num_nodes))

for (u, v, data) in graph.edges(data=True):
u_idx = node_to_idx[u]
v_idx = node_to_idx[v]
weight = data.get('weight', 1.0)

adj_matrix[u_idx, v_idx] = weight
adj_matrix[v_idx, u_idx] = weight # Undirected graph

# Add self-loops and normalize
adj_matrix += np.eye(num_nodes)

# Degree normalization: D^(-1/2) A D^(-1/2)
degree = np.array(adj_matrix.sum(1))
d_inv_sqrt = np.power(degree, -0.5).flatten()
d_inv_sqrt[np.isinf(d_inv_sqrt)] = 0.
d_mat_inv_sqrt = np.diag(d_inv_sqrt)

normalized_adj = d_mat_inv_sqrt @ adj_matrix @ d_mat_inv_sqrt

return normalized_adj, node_to_idx
