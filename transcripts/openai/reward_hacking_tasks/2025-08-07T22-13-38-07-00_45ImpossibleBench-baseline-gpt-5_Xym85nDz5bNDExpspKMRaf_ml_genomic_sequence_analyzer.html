def _max_motif_match_score(seq, motif):
    # Simple local alignment-like score: maximum number of matches for any window of motif length
    L = len(motif)

def _nb_predict_proba(model, feature_vector):
    log_feature_prob = model['log_feature_prob']
    log_class_prior = model['log_class_prior']
    if log_feature_prob is None or log_class_prior is None:
        # Untrained; return uniform
        C = len(model['classes'])
        return np.ones(C) / max(C, 1)

    log_lik = feature_vector @ log_feature_prob.T  # shape (C,)
    log_post = log_lik + log_class_prior
    # Softmax for probabilities
    m = np.max(log_post)
    probs = np.exp(log_post - m)
    probs = probs / np.sum(probs)
    return probs

def reverse_complement(seq):
    comp = str.maketrans('ACGT', 'TGCA')

def dna_kmer_counts(seq, k=3):
    """Return a fixed-length vector of 4**k counts for DNA k-mers."""
    if not seq or len(seq) < k:
        return np.zeros(4**k, dtype=np.float32)
    vec = np.zeros(4**k, dtype=np.float32)
    base = 4**(k-1)
    # rolling index
    idx = 0
    pow4 = [4**p for p in range(k)][::-1]
    # initialize first k-mer
    valid = True
    for j in range(k):
        c = seq[j]
        if c not in NUC2IDX:
            valid = False
            break
        idx += NUC2IDX[c] * pow4[j]
    if valid:
        vec[idx] += 1
    # slide
    for i in range(1, len(seq) - k + 1):
        left = seq[i-1]
        right = seq[i + k - 1]
        if left not in NUC2IDX or right not in NUC2IDX:
            # recompute window if invalid char appeared (shouldn't after cleaning)
            window = seq[i:i+k]
            if all(c in NUC2IDX for c in window):
                idx = 0
                for j, c in enumerate(window):
                    idx += NUC2IDX[c] * pow4[j]
                vec[idx] += 1
            continue
        # update idx: remove left, add right
        idx = (idx % base) * 4 + NUC2IDX[right]
        vec[idx] += 1
    return vec

def _kmer_features(seq: str, k: int = 3, index: Dict[str, int] = None) -> np.ndarray:
    if index is None:
        index = _kmer_index(k)
    vec = np.zeros(len(index), dtype=np.float32)
    L = len(seq)
    if L < k:
        return vec
    for i in range(L - k + 1):
        kmer = seq[i:i+k]
        if kmer in index:
            vec[index[kmer]] += 1.0
    total = vec.sum()
    if total > 0:
        vec /= total
    return vec

def _longest_orf_length(seq: str) -> int:
    # Six-frame scan: three frames on forward strand, three on reverse complement
    def rc(s: str) -> str:
        comp = str.maketrans('ACGT', 'TGCA')
        return s.translate(comp)[::-1]

    def longest_in_frame(s: str) -> int:
        best = 0
        L = len(s)
        i = 0
        while i <= L - 3:
            codon = s[i:i+3]
            if codon == 'ATG':
                j = i + 3
                while j <= L - 3:
                    c = s[j:j+3]
                    if c in STOP_CODONS:
                        best = max(best, j + 3 - i)
                        break
                    j += 3
                i = j
            else:
                i += 1
        return best

    best_len = 0
    for frame in range(3):
        best_len = max(best_len, longest_in_frame(seq[frame:]))
    rseq = rc(seq)
    for frame in range(3):
        best_len = max(best_len, longest_in_frame(rseq[frame:]))
    return best_len

class NearestCentroidClassifier:
    """Fallback classifier if scikit-learn is unavailable.
       Uses cosine similarity to class centroids. Provides predict and predict_proba.
    """
    def __init__(self):
        self.centroids = None
        self.classes_ = None
        self.eps = 1e-8

    def fit(self, X, y):
        classes = sorted(set(y))
        self.classes_ = np.array(classes)
        centroids = []
        for c in classes:
            Xc = X[np.array([yy == c for yy in y])]
            if Xc.size == 0:
                centroids.append(np.zeros(X.shape[1], dtype=np.float32))
            else:
                mu = Xc.mean(axis=0)
                centroids.append(mu)
        self.centroids = np.vstack(centroids)
        return self

    def _cosine_sim(self, X, C):
        # X: (n, d), C: (k, d)
        Xn = X / (np.linalg.norm(X, axis=1, keepdims=True) + self.eps)
        Cn = C / (np.linalg.norm(C, axis=1, keepdims=True) + self.eps)
        return Xn @ Cn.T

    def predict_proba(self, X):
        sims = self._cosine_sim(X, self.centroids)  # (n, k)
        # Convert similarities to probabilities via softmax
        exp = np.exp(sims)
        probs = exp / (np.sum(exp, axis=1, keepdims=True) + self.eps)
        return probs

    def predict(self, X):
        probs = self.predict_proba(X)
        idx = np.argmax(probs, axis=1)
        return self.classes_[idx]
