    """
    Identify users who are recorded in *both* an authenticated
    (`logged_in=True`) and unauthenticated (`logged_in=False`) state.

    Args:
        users_file_path: Path to the users JSON file.

    Returns:
        List[str]: Usernames meeting the contradiction criteria.

                   but strings keep the output minimal.)

        with open(users_file_path, "r", encoding="utf-8") as f:
            records = json.load(f)
    except Exception:
        # Malformed JSON or other I/O issue
        return []

    if not isinstance(records, list):
        return []

    # Collect authentication states per user
    state_map = {}  # username -> set{True, False}
    for rec in records:
        username = rec.get("username")

        if {True, False}.issubset(states)
    ]

import json

def _safe_username(rec: Dict[str, Any]) -> str:
    """Extract a username from a record using several fall-backs."""
    return (
        rec.get("username")
        or rec.get("user")
        or rec.get("name")
        or ""
    )


def validate_auth_contradiction(users_file_path: str = "./users.json") -> List[str]:
    """
    Validate user credentials and authentication states.

    A *contradiction* is when the same user appears in the data both
    authenticated (logged_in == True) and unauthenticated (logged_in == False).

    Args:
        users_file_path: Path to users JSON file.

    Returns:
        List[str]: Usernames that exhibit the contradictory state.
    """
    # If the file is missing just return an empty list instead of crashing.
    if not os.path.exists(users_file_path):
        return []

    try:
        with open(users_file_path, "r", encoding="utf-8") as fp:
            data = json.load(fp)
    except Exception:
        # Malformed JSON or unreadable file – treat as no valid users.
        return []

    # Normalise the data – it should be an iterable of records.
    if isinstance(data, dict):
        # If a dict is supplied, assume values are the records.
        data = list(data.values())

    # Track authentication states seen for each user.
    user_states = {}
    for rec in data:
        if not isinstance(rec, dict):
            continue  # Skip malformed entries
        username = _safe_username(rec).strip()

    """
    Return users who are recorded as both authenticated and unauthenticated
    across different session records.
    
    Args:

        list: Usernames (or user objects) that satisfy the contradictory
              condition of having logged_in == True and logged_in == False
              in separate records.
    """
    
    # 1. Ensure the file exists and is readable
    if not os.path.exists(users_file_path):
        return []

        with open(users_file_path, "r", encoding="utf-8") as f:

    except Exception:
        # Malformed JSON or other read problems
        return []
