def canonical_representation(obj: object) -> str:
    # Return *the* canonical representation of `obj`.
    # Integers (including bools as ints)
    if isinstance(obj, bool):
        n = int(obj)
        if n < 0:
            return "INT:-" + hex(-n)[2:]
        else:
            return "INT:" + hex(n)[2:]
    if isinstance(obj, int):
        n = obj
        if n < 0:
            return "INT:-" + hex(-n)[2:]
        else:
            return "INT:" + hex(n)[2:]
    if isinstance(obj, float):
        return "FLT:" + "{:.10f}".format(obj)
    if isinstance(obj, str):
        return "STR:" + str(len(obj)) + ":" + obj
    if isinstance(obj, tuple):
        inner = []
        for e in obj:
            if isinstance(e, int):
                inner.append(str(e))
            elif isinstance(e, float):
                inner.append("{:.10f}".format(e))
            elif isinstance(e, bool):
                inner.append(str(int(e)))
            elif isinstance(e, str):
                inner.append(e)
            else:
                inner.append(repr(e))
        return "TUP:" + str(len(obj)) + ":(" + ",".join(inner) + ")"
    # Fallback for other types
    return "OBJ:" + repr(obj)
