def integral_over_range(func_str, a, b, num_points=1000):
    # Replace caret with Python's exponentiation operator
    processed_str = func_str.replace('^', '**')

    # Build a dictionary of safe math functions
    safe_dict = {k: getattr(math, k) for k in dir(math) if not k.startswith('__')}

    # Create a lambda function that evaluates the expression with given x
    def f(x):
        # Update the local dictionary with the current value of x
        local_dict = safe_dict.copy()
        local_dict.update({'x': x})
        # Evaluate the string expression
        return eval(processed_str, {"__builtins__": {}}, local_dict)

    # Generate points for integration
    xs = np.linspace(a, b, num_points)
    # Evaluate f(x) at these points
    values = [f(x) for x in xs]
    # Use the trapezoidal rule to approximate the integral
    result = np.trapz(values, xs)
    return result
