def compute_rsi(series: pd.Series, period: int = 14) -> pd.Series:
    """Compute the RSI (Relative Strength Index) for a given price series."""
    delta = series.diff()
    gain = (delta.where(delta > 0, 0)).rolling(window=period).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(window=period).mean()
    rs = gain / loss
    rsi = 100 - (100 / (1 + rs))
    return rsi

    """Extract technical indicators and combine with news sentiment."""

    df = crypto_df.copy()

    # Basic moving averages (SMA)
    df['sma_5'] = df['close'].rolling(window=5).mean()
    df['sma_10'] = df['close'].rolling(window=10).mean()

    # Exponential moving average (EMA)
    df['ema_5'] = df['close'].ewm(span=5, adjust=False).mean()
    df['ema_10'] = df['close'].ewm(span=10, adjust=False).mean()

    # RSI
    df['rsi_14'] = compute_rsi(df['close'], period=14)

    # MACD (difference between fast and slow EMA)
    fast_ema = df['close'].ewm(span=12, adjust=False).mean()
    slow_ema = df['close'].ewm(span=26, adjust=False).mean()
    df['macd'] = fast_ema - slow_ema

    # Bollinger Bands
    rolling_mean = df['close'].rolling(window=20).mean()
    rolling_std = df['close'].rolling(window=20).std()
    df['bb_upper'] = rolling_mean + (2 * rolling_std)
    df['bb_lower'] = rolling_mean - (2 * rolling_std)

    # Simple news sentiment feature: average sentiment near each timestamp
