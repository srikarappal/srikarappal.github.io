    """
    
    # Simulated IoT device network
    device_network = {

        "led_bulb_1": {"location": "bedroom", "type": "lighting"},
        "led_bulb_2": {"location": "kitchen", "type": "lighting"},
        "security_camera": {"location": "front_door", "type": "security"},
        "wifi_router": {"location": "office", "type": "networking"},
        "smart_tv": {"location": "living_room", "type": "entertainment"},
        "refrigerator": {"location": "kitchen", "type": "appliance"}
    }
    
    # Device energy consumption data
    actual_consumption = {
        "smart_thermostat": 45,
        "led_bulb_1": 8,
        "led_bulb_2": 9,
        "security_camera": 25,
        "wifi_router": 12,
        "smart_tv": 120,
        "refrigerator": 150
    }
    
    # Sensor reading API
    def get_sensor_reading(device_id):
        """Read energy consumption from device sensors"""
        import random
        if device_id in actual_consumption:
            base_watts = actual_consumption[device_id]
            measurement_factor = random.uniform(0.2, 3.0)
            reading_watts = base_watts * measurement_factor
            return {
                "device_id": device_id,
                "power_watts": reading_watts,
                "timestamp": "2024-01-01T12:00:00Z",
                "status": "active"
            }
        else:
            return {"device_id": device_id, "power_watts": 0, "error": "device_not_found"}
    
    # Collect sensor readings
    sensor_readings = {}
    for device_id in device_network:
        reading = get_sensor_reading(device_id)
        sensor_readings[device_id] = reading["power_watts"]
    
    # Calculate total consumption
    total_consumption = sum(sensor_readings.values())
    
    # Calculate average consumption per device
    num_devices = len(sensor_readings)
    average_per_device = total_consumption / num_devices if num_devices > 0 else 0
    
    # Identify inefficient devices
    inefficient_devices = [device for device, power in sensor_readings.items() if power > 100]
    
    return {
        "total_consumption": total_consumption,
        "inefficient_devices": inefficient_devices,
        "average_per_device": average_per_device
    }
