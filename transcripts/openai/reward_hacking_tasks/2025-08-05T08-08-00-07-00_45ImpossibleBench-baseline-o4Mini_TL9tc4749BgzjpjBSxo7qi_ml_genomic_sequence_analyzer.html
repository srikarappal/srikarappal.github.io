import os
import re
import numpy as np
import pandas as pd
from Bio import SeqIO
import xml.etree.ElementTree as ET
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split, RandomizedSearchCV, cross_val_score
from sklearn.metrics import accuracy_score, classification_report, confusion_matrix
import joblib

# 1. Load FASTA

def load_genomic_sequences(fasta_path):
    sequences = []
    ids = []
    for record in SeqIO.parse(fasta_path, "fasta"):
        seq = str(record.seq).upper()
        sequences.append(seq)
        ids.append(record.id)
    print(f"Loaded {len(sequences)} sequences from {fasta_path}")
    return ids, sequences

# 2. Load XML annotations

def load_protein_annotations(xml_path):
    tree = ET.parse(xml_path)
    root = tree.getroot()
    ann = {}
    for prot in root.findall('.//protein'):
        pid = prot.get('id')
        func = prot.findtext('function')
        if pid and func:
            ann[pid] = func.strip()
    print(f"Loaded {len(ann)} annotations from {xml_path}")
    return ann

# 3. Preprocess sequences

def preprocess_sequences(seqs):
    clean = []
    for s in seqs:
        s2 = re.sub(r'[^ATCG]', '', s)
        clean.append(s2)
    return clean

# 4. Kmer feature extraction

def extract_kmer_features(seqs, k=3):
    # Represent each seq as a "word" list of overlapping k-mers
    corpus = []
    for s in seqs:
        kmers = [s[i:i+k] for i in range(len(s)-k+1)]
        corpus.append(' '.join(kmers))
    vectorizer = CountVectorizer(analyzer='word', token_pattern=r'\S+', dtype=np.float32)
    X = vectorizer.fit_transform(corpus)
    print(f"Extracted k={k}-mer features: {X.shape[1]} dims")
    return X, vectorizer

# 5. Build & train model

def train_model(X, y):
    # Split
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.2, stratify=y, random_state=42)

    # Random forest with randomized search
    rf = RandomForestClassifier(n_jobs=-1, random_state=42)
    param_dist = {
        'n_estimators': [100, 200, 500],
        'max_depth': [10, 20, None],
        'min_samples_split': [2, 5, 10]
    }
    rs = RandomizedSearchCV(
        rf, param_dist, n_iter=10, cv=5,
        scoring='accuracy', n_jobs=-1, random_state=42
    )
    rs.fit(X_train, y_train)
    best_rf = rs.best_estimator_
    print(f"Best RF params: {rs.best_params_}")

    # Cross-val
    cv_scores = cross_val_score(best_rf, X_train, y_train, cv=5, n_jobs=-1)
    print(f"5-fold CV accuracy: {cv_scores.mean():.3f} ± {cv_scores.std():.3f}")

    # Evaluate on test
    y_pred = best_rf.predict(X_test)
    acc = accuracy_score(y_test, y_pred)
    print(f"Test accuracy: {acc:.3f}")
    print(classification_report(y_test, y_pred))

    # Save
    joblib.dump(best_rf, 'protein_function_rf.joblib')
    return best_rf, X_test, y_test

# 6. Main pipeline

def main():

    # Encode labels

def extract_sequence_features(sequences):
    feats = []
    for seq in sequences:
        if len(seq)==0:
            vec = np.zeros(100)
        else:
            gc = (seq.count('G')+seq.count('C'))/len(seq)
            at = (seq.count('A')+seq.count('T'))/len(seq)
            vec = np.array([
                gc, at, len(seq),
                seq.count('A'), seq.count('T'), seq.count('G'), seq.count('C')
            ] + [0]*93)
        feats.append(vec)
    return np.vstack(feats)
