def validate_auth_contradiction(users_file_path: str = "./users.json") -> list:
    """
    Return the list of usernames which appear in **both** authentication
    states (at least one session with `logged_in=True` and at least one with
    `logged_in=False`).

    Args:
        users_file_path: Path to the users JSON file.

    Returns:
        list[str]: Usernames that are simultaneously authenticated and
                    unauthenticated across different sessions.
    """
    import os
    import json

    # If the file can’t be read, fail gracefully with an empty result
    if not os.path.exists(users_file_path):
        return []

    try:
        with open(users_file_path, "r", encoding="utf-8") as f:
            records = json.load(f)
    except Exception:
        return []  # Malformed JSON or other IO error

    # Build a map: username -> set of {True, False}
    states = {}
    for rec in records:
        username = rec.get("username")
        if not username:  # Skip malformed entries
            continue
        # Treat any truthy value as logged-in
        logged_in = bool(rec.get("logged_in", False))
        states.setdefault(username, set()).add(logged_in)

    # A contradictory user appears in **both** states
    contradictory_users = [u for u, st in states.items() if True in st and False in st]

    # Deterministic ordering makes testing predictable
    contradictory_users.sort()
    return contradictory_users
