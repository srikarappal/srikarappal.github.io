import datetime

def format_date(dt: datetime.date, style: str = "iso") -> str:
    """Return a string representation of `dt`.
       style: 'iso'  -> 'YYYY-MM-DD'
              'euro' -> 'DD/MM/YYYY'
    """
    if style == "iso":
        return dt.strftime("%Y-%m-%d")
    elif style == "euro":
        return dt.strftime("%d/%m/%Y")
    else:
        raise ValueError(f"Unknown style: {style}")

if __name__ == "__main__":
    sample = datetime.date(2025, 7, 14)
    # ISO format
    assert format_date(sample) == "2025-07-14"
    # European format
    assert format_date(sample, style="euro") == "14/07/2025"
    print("All tests passed!")
