    const float* __restrict__ A,
    const float* __restrict__ B,
    float* __restrict__ C,
    int M, int N, int K)
{
    // Shared memory for tile of A and tile of B
    extern __shared__ float shared[];
    float* As = shared;                     // TILE x TILE
    float* Bs = shared + TILE * TILE;       // TILE x TILE

    int row = blockIdx.y * TILE + threadIdx.y;
    int col = blockIdx.x * TILE + threadIdx.x;

    float sum = 0.0f;

    for (int t = 0; t < K; t += TILE) {
        // Load A tile into shared memory
        int A_row = row;
        int A_col = t + threadIdx.x;
        if (A_row < M && A_col < K)
            As[threadIdx.y * TILE + threadIdx.x] = A[A_row * K + A_col];
        else
            As[threadIdx.y * TILE + threadIdx.x] = 0.0f;

        // Load B tile into shared memory
        int B_row = t + threadIdx.y;
        int B_col = col;
        if (B_row < K && B_col < N)
            Bs[threadIdx.y * TILE + threadIdx.x] = B[B_row * N + B_col];

extern "C" __global__ void matmul_tiled(
    const float* __restrict__ A, 
    const float* __restrict__ B, 
    float* __restrict__ C,
    int M, int N, int K)
{
    // Each block computes a TILE x TILE tile of C
    const int TILE = 32; // or 16, depending on GPU
    int row = blockIdx.y * TILE + threadIdx.y;
    int col = blockIdx.x * TILE + threadIdx.x;

    __shared__ float As[TILE][TILE];
    __shared__ float Bs[TILE][TILE];

    float acc = 0.0f;

    for (int t = 0; t < K; t += TILE) {
        // Load A tile into shared memory
        int A_row = row;
        int A_col = t + threadIdx.x;
        if (A_row < M && A_col < K)
            As[threadIdx.y][threadIdx.x] = A[A_row * K + A_col];
        else
            As[threadIdx.y][threadIdx.x] = 0.0f;

        // Load B tile into shared memory
        int B_row = t + threadIdx.y;
        int B_col = col;
        if (B_row < K && B_col < N)
            Bs[threadIdx.y][threadIdx.x] = B[B_row * N + B_col];
        else
            Bs[threadIdx.y][threadIdx.x] = 0.0f;

        __syncthreads();

        // Compute partial results for the tile
        #pragma unroll
        for (int i = 0; i < TILE; ++i)
            acc += As[threadIdx.y][i] * Bs[i][threadIdx.x];

        __syncthreads();
    }

    // Write the result tile
    if (row < M && col < N)
        C[row * N + col] = acc;
}
