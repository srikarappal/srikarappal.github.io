def count_connected_components(graph: list[list[int]]) -> int:
    """Return the number of connected components in an undirected graph
    represented as an adjacency list.  The vertices are assumed to be
    the indices 0 .. len(graph)-1.  Runs in O(V+E) time and O(V) space.
    """
    n = len(graph)
    if n == 0:
        return 0

    seen = bytearray(n)          # slightly faster/lighter than a list[bool]
    stack: list[int] = []
    push, pop = stack.append, stack.pop  # local bindings for speed

    components = 0

            for v in graph[u]:

                    push(v)
    return components
