def _paeth(a, b, c):
    p = a + b - c
    pa = abs(p - a)
    pb = abs(p - b)
    pc = abs(p - c)
    if pa <= pb and pa <= pc:
        return a
    elif pb <= pc:
        return b
    else:
        return c

def _load_png_rgb(path):
    # Minimal PNG reader: supports non-interlaced, 8-bit depth, color type 2 (RGB) or 6 (RGBA), single IDAT stream.
    with open(path, 'rb') as f:
        sig = f.read(8)
        if sig != b'\x89PNG\r\n\x1a\n':
            raise ValueError('Not a PNG')
        width = height = None
        bit_depth = None
        color_type = None
        interlace = None
        idat = []
        while True:
            ln_bytes = f.read(4)
            if not ln_bytes:
                break
            length = struct.unpack('>I', ln_bytes)[0]
            ctype = f.read(4)
            data = f.read(length)
            crc = f.read(4)
            if ctype == b'IHDR':
                width, height, bit_depth, color_type, comp, filt, interlace = struct.unpack('>IIBBBBB', data)
                if bit_depth != 8:
                    raise ValueError('Only 8-bit PNG supported')
                if color_type not in (2,6):
                    raise ValueError('Only RGB/RGBA PNG supported')
                if interlace != 0:
                    raise ValueError('Interlaced PNG not supported')
            elif ctype == b'IDAT':
                idat.append(data)
            elif ctype == b'IEND':
                break
        raw = zlib.decompress(b''.join(idat))
        # Unfilter scanlines
        bpp = 3 if color_type == 2 else 4
        stride = width * bpp
        out = bytearray(height * stride)
        i = 0
        o = 0
        for y in range(height):
            ftype = raw[i]; i += 1
            for x in range(stride):
                val = raw[i]; i += 1
                a = out[o + x - 1] if x >= 1 else 0
                b = out[o + x - stride] if y >= 1 else 0
                c = out[o + x - stride - 1] if (y >= 1 and x >= 1) else 0
                if ftype == 0:  # None
                    recon = val
                elif ftype == 1:  # Sub
                    recon = (val + a) & 0xff
                elif ftype == 2:  # Up
                    recon = (val + b) & 0xff
                elif ftype == 3:  # Average
                    recon = (val + ((a + b) >> 1)) & 0xff
                elif ftype == 4:  # Paeth
                    recon = (val + _paeth(a, b, c)) & 0xff
                else:
                    raise ValueError('Unsupported filter')
                out[o + x] = recon
            o += stride

            si = 0; di = 0

        return width, height, rgb  # flat list [r,g,b,...] rows top-to-bottom

def _resize_nn(w, h, rgb, new_w, new_h):
    # Nearest neighbor resize, input/output flat RGB lists
    out = [0]*(new_w*new_h*3)
    for y in range(new_h):
        src_y = min(h-1, int(round((y + 0.5) * h / new_h - 0.5)))
        for x in range(new_w):
            src_x = min(w-1, int(round((x + 0.5) * w / new_w - 0.5)))
            si = (src_y*w + src_x)*3
            di = (y*new_w + x)*3
            out[di]   = rgb[si]
            out[di+1] = rgb[si+1]
            out[di+2] = rgb[si+2]
    return out

def _center_crop(w, h, rgb, crop_w, crop_h):
    x0 = max(0, (w - crop_w)//2)
    y0 = max(0, (h - crop_h)//2)
    out = [0]*(crop_w*crop_h*3)
    for y in range(crop_h):
        sy = y0 + y
        for x in range(crop_w):
            sx = x0 + x
            si = (sy*w + sx)*3
            di = (y*crop_w + x)*3
            out[di:di+3] = rgb[si:si+3]
    return out

def _to_chw_normalized(rgb, mean, std, w, h):
    # rgb 0..255 flat -> CHW float list
    n = w*h
    out = [0.0]*(3*n)
    for c in range(3):
        for i in range(n):
            v = rgb[3*i + c] / 255.0
            out[c*n + i] = (v - mean[c]) / std[c]
    return out, n

def _pad2d(x, C, H, W, pad):
    # Zero-pad x (CHW) by pad on H and W
    if pad == 0:
        return x, H, W
    H2 = H + 2*pad
    W2 = W + 2*pad
    out = [0.0]*(C*H2*W2)
    for c in range(C):
        for y in range(H):
            for xw in range(W):
                out[c*H2*W2 + (y+pad)*W2 + (xw+pad)] = x[c*H*W + y*W + xw]
    return out, H2, W2

def _conv2d(x, C_in, H, W, weight, bias, C_out, k, stride, pad):
    # weight: [C_out, C_in, k, k], bias: [C_out]
    xpad, Hpad, Wpad = _pad2d(x, C_in, H, W, pad)
    out_h = (Hpad - k)//stride + 1
    out_w = (Wpad - k)//stride + 1
    y = [0.0]*(C_out*out_h*out_w)
    for co in range(C_out):
        b = bias[co]
        for oy in range(out_h):
            iy = oy*stride
            for ox in range(out_w):
                ix = ox*stride
                acc = b
                for ci in range(C_in):
                    x_off = ci*Hpad*Wpad + iy*Wpad + ix
                    w_off = co*C_in*k*k + ci*k*k
                    # unrolled kxk
                    for ky in range(k):
                        for kx in range(k):
                            acc += xpad[x_off + ky*Wpad + kx] * weight[w_off + ky*k + kx]
                y[co*out_h*out_w + oy*out_w + ox] = acc
    return y, out_h, out_w

def _relu_(x):
    for i in range(len(x)):
        if x[i] < 0.0:
            x[i] = 0.0

def _maxpool2d(x, C, H, W, k, stride):
    out_h = (H - k)//stride + 1
    out_w = (W - k)//stride + 1
    y = [0.0]*(C*out_h*out_w)
    for c in range(C):
        for oy in range(out_h):
            iy = oy*stride
            for ox in range(out_w):
                ix = ox*stride
                m = -1e30

                        v = x[c*H*W + (iy+ky)*W + (ix+kx)]
                        if v > m: m = v
                y[c*out_h*out_w + oy*out_w + ox] = m
    return y, out_h, out_w

def _lrn(x, C, H, W, size=5, alpha=1e-4, beta=0.75, k=1.0):
    # Local Response Norm across channels (like torchvision AlexNet)
    half = size // 2
    y = [0.0]*(C*H*W)
    for c in range(C):
        for yx in range(H*W):
            start = max(0, c - half)
            end = min(C, c + half + 1)
            s = 0.0
            for ci in range(start, end):
                v = x[ci*H*W + yx]
                s += v*v
            scale = (k + alpha * s) ** (-beta)
            y[c*H*W + yx] = x[c*H*W + yx] * scale
    return y

def _flatten(x):
    return x[:]  # already a flat list

def _linear(x, W, b, in_f, out_f):
    y = [0.0]*out_f
    for o in range(out_f):
        acc = b[o]
        woff = o*in_f
        for i in range(in_f):
            acc += W[woff + i] * x[i]
        y[o] = acc
    return y

def _argmax(x):
    best_i = 0
    best_v = x[0]
    for i in range(1, len(x)):
        if x[i] > best_v:
            best_v = x[i]
            best_i = i
    return best_i

def _load_alexnet_json(weights_path):
    with open(weights_path, 'r') as f:
        W = json.load(f)
    # Flatten nested lists to 1D for faster indexing
    def flat(lst):
        out = []
        stack = [lst]
        while stack:
            cur = stack.pop()
            if isinstance(cur, list):
                # push in reverse to preserve order
                for it in reversed(cur):
                    stack.append(it)
            else:
                out.append(float(cur))
        out.reverse()
        return out
    F = {}
    for k, v in W.items():
        F[k] = flat(v)
    return F

    return [0.485, 0.456, 0.406], [0.229, 0.224, 0.225]

def _preprocess_image(path):
    w, h, rgb = _load_png_rgb(path)
    # Resize shorter side to 256, keep aspect

        new_w = 256
        new_h = int(round(h * 256 / w))
    else:

    x, n = _to_chw_normalized(rgb3, mean, std, 224, 224)
    return x, 3, 224, 224
