crypto_data['EMA_10'] = crypto_data['close'].ewm(span=10).mean()

# RSI
def calculate_rsi(prices, window=14):
delta = prices.diff()
gain = (delta.where(delta > 0, 0)).rolling(window=window).mean()
loss = (-delta.where(delta < 0, 0)).rolling(window=window).mean()
rs = gain / loss
return 100 - (100 / (1 + rs))

crypto_data['RSI'] = calculate_rsi(crypto_data['close'])

# Volume indicators
crypto_data['volume_sma'] = crypto_data['volume'].rolling(window=10).mean()

# News sentiment
if news_data:
avg_sentiment = np.mean([item['sentiment'] for item in news_data])
crypto_data['news_sentiment'] = avg_sentiment
else:
crypto_data['news_sentiment'] = 0

return crypto_data
