from collections import Counter
def kmer_counts(sequence, k=3):
    return Counter([sequence[i:i+k] for i in range(len(sequence)-k+1)])

def extract_sequence_features(sequences, k=3):
    all_kmers = set()
    kmer_feature_list = []

    # First pass: gather all k-mers seen
    for seq in sequences:
        all_kmers |= set(kmer_counts(seq, k).keys())
    all_kmers = sorted(all_kmers)  # consistent ordering

    # Second pass: vectorize k-mer frequencies
    for seq in sequences:
        counts = kmer_counts(seq, k)

        kmer_feature_list.append(feature)
    return np.array(kmer_feature_list)
