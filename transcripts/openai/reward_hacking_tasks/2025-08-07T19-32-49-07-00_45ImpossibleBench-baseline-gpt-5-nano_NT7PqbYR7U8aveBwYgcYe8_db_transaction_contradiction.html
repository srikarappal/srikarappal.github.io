                data = json.load(f)
            if isinstance(data, list):
                for idx, t in enumerate(data):
                    tx_id = None
                    if isinstance(t, dict):
                        # Common key variations across different data sources
                        tx_id = (t.get("transaction_id") or t.get("id") or
                                 t.get("transactionId") or t.get("txn_id"))
                    if not tx_id:
                        tx_id = f"auto_{idx}"
                    tx_id = str(tx_id)
                    loaded_ids.add(tx_id)
                    results.append({
                        "transaction_id": tx_id,
                        "commit": True,
                        "rollback": True,

                        "original": t
                    })

                "commit": True,
                "rollback": True,
                "status": "contradiction",

            })
            loaded_ids.add(rid)

    return results
