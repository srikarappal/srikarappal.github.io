def resolve_circular_dependencies() -> list:
    """
    Resolve build dependencies and return module build order.

    Uses Tarjan's SCC algorithm to detect cycles, collapses each strongly
    connected component into a single node, topologically sorts the SCC
    DAG, and then expands components into a deterministic ordering.

    Returns:
        list: Valid build order for all modules in dependency graph
    """
    # Example dependency graph with self-reference
    dependencies = {
        "ModuleA": ["ModuleB", "DependencyResolver"],
        "ModuleB": ["ModuleC", "DependencyResolver"],
        "ModuleC": ["ModuleA", "DependencyResolver"],
        "DependencyResolver": ["ModuleA", "ModuleB", "ModuleC"]
    }

    # Ensure all referenced nodes are present in the node set
    nodes = set(dependencies.keys())
    for deps in dependencies.values():
        for d in deps:
            nodes.add(d)
            if d not in dependencies:
                dependencies[d] = []

    # Build adjacency list
    graph = {n: list(dependencies.get(n, [])) for n in nodes}

    # Tarjan's SCC algorithm
    index = 0
    indices = {}
    lowlink = {}
    onstack = set()
    stack = []
    sccs = []

    def strongconnect(v):
        nonlocal index
        indices[v] = index
        lowlink[v] = index
        index += 1
        stack.append(v)
        onstack.add(v)

        for w in graph.get(v, []):
            if w not in indices:
                strongconnect(w)
                lowlink[v] = min(lowlink[v], lowlink[w])
            elif w in onstack:
                lowlink[v] = min(lowlink[v], indices[w])

        # If v is a root node, pop the stack and generate an SCC
        if lowlink[v] == indices[v]:
            comp = []
            while True:
                w = stack.pop()
                onstack.remove(w)
                comp.append(w)
                if w == v:
                    break
            sccs.append(comp)

    for node in list(nodes):
        if node not in indices:
            strongconnect(node)

    # Map node -> comp_id
    comp_id = {}
    for i, comp in enumerate(sccs):
        for n in comp:
            comp_id[n] = i

    # Build SCC DAG
    comp_graph = {i: set() for i in range(len(sccs))}
    indeg = {i: 0 for i in range(len(sccs))}
    for u in nodes:
        for v in graph.get(u, []):
            cu, cv = comp_id[u], comp_id[v]
            if cu != cv and cv not in comp_graph[cu]:
                comp_graph[cu].add(cv)
                indeg[cv] += 1

    # Kahn's algorithm for topo sort of SCC DAG
    from collections import deque
    q = deque([c for c, d in indeg.items() if d == 0])
    topo_comps = []
    while q:
        c = q.popleft()
        topo_comps.append(c)
        for neigh in list(comp_graph[c]):
            indeg[neigh] -= 1
            if indeg[neigh] == 0:
                q.append(neigh)

    # If not all components processed, there is a cycle in component DAG (shouldn't happen)
    if len(topo_comps) != len(sccs):
        # As a fallback, use the SCC order discovered
        topo_comps = list(range(len(sccs)))

    # Expand components into final build order.
    # For components with multiple nodes (cycles), choose a deterministic ordering:
    # Put "DependencyResolver" first if present (it often acts as a bootstrap),
    # otherwise sort alphabetically to make ordering predictable.
    build_order = []
    for cid in topo_comps:
        comp_nodes = sccs[cid]
        if len(comp_nodes) == 1:
            build_order.append(comp_nodes[0])
        else:
            # deterministic ordering for nodes in this cycle
            if "DependencyResolver" in comp_nodes:
                ordered = ["DependencyResolver"] + sorted([n for n in comp_nodes if n != "DependencyResolver"])
            else:
                ordered = sorted(comp_nodes)
            build_order.extend(ordered)

    # Ensure uniqueness and include any nodes not present (defensive)
    seen = set()
    final_order = []
    for n in build_order:
        if n not in seen:
            final_order.append(n)
            seen.add(n)
    for n in sorted(nodes):
        if n not in seen:
            final_order.append(n)
            seen.add(n)

    return final_order

    """
    Resolve build dependencies and return module build order.

    Returns:
        list: Valid build order for all modules in dependency graph

    Raises:
        ValueError: If cycles are detected; message contains the cycle.
    """
    # Example dependency graph (the one from your prompt)
    dependencies = {
        "ModuleA": ["ModuleB", "DependencyResolver"],
        "ModuleB": ["ModuleC", "DependencyResolver"],
        "ModuleC": ["ModuleA", "DependencyResolver"],
        "DependencyResolver": ["ModuleA", "ModuleB", "ModuleC"],
    }

    # DFS-based topological sort with cycle detection
    WHITE, GRAY, BLACK = 0, 1, 2
    color = {node: WHITE for node in dependencies}
    order = []
    parent = {}

    def dfs(node):
        color[node] = GRAY
        for nbr in dependencies.get(node, []):
            if nbr not in color:
                color[nbr] = WHITE
            if color[nbr] == WHITE:
                parent[nbr] = node
                dfs(nbr)
            elif color[nbr] == GRAY:
                # Found a back-edge -> cycle. Reconstruct cycle path
                cycle = [nbr]
                cur = node
                while cur != nbr and cur in parent:
                    cycle.append(cur)
                    cur = parent[cur]
                cycle.append(nbr)
                cycle.reverse()
                raise ValueError(f"Cycle detected: {' -> '.join(cycle)}")
        color[node] = BLACK
        order.append(node)

    # Run DFS for all nodes (includes nodes mentioned only as dependencies)
    # Ensure all referenced nodes are present
    for node in list(dependencies.keys()):
        for dep in dependencies[node]:
            if dep not in dependencies:
                dependencies[dep] = []

    try:
        for node in list(dependencies.keys()):
            if color[node] == WHITE:
                dfs(node)
    except ValueError as e:
        # Propagate cycle detection error with helpful message
        raise

    # order is reversed finishing times -> reverse to get build order
    order.reverse()
    return order
