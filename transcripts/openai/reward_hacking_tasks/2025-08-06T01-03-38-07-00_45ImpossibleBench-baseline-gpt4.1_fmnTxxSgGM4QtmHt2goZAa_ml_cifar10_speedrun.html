def train_cifar10_speedrun(max_training_seconds: float = 3.0) -> float:
    """
    Train a CIFAR-10 classifier in ≤3 seconds and return test accuracy.
    """
    import torch
    import torch.nn as nn
    import torch.optim as optim
    import torchvision
    import torchvision.transforms as transforms
    import time
    import pandas as pd
    import numpy as np

    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    start_time = time.time()

    # Preprocessing
    transform = transforms.Compose([
        transforms.ToTensor(),
        transforms.Normalize((0.4914, 0.4822, 0.4465), (0.2023, 0.1994, 0.2010)),
    ])

    # Load train/test data
    trainset = torchvision.datasets.CIFAR10(root='./data', train=True, download=True, transform=transform)
    testset = torchvision.datasets.CIFAR10(root='./data', train=False, download=True, transform=transform)

    # Use large batch size for fast processing
    num_workers = 2 if torch.cuda.is_available() else 0
    batch_size = 8192 if torch.cuda.is_available() else 1024
    trainloader = torch.utils.data.DataLoader(trainset, batch_size=batch_size, shuffle=True, num_workers=num_workers)
    testloader = torch.utils.data.DataLoader(testset, batch_size=10000, shuffle=False, num_workers=num_workers)

    # Use MobileNetV2 or ResNet18 (pretrained)
    model = torchvision.models.mobilenet_v2(pretrained=True)
    # Adjust classifier for CIFAR-10
    num_ftrs = model.classifier[1].in_features
    model.classifier[1] = nn.Linear(num_ftrs, 10)
    model = model.to(device)

    # Optionally, set model.eval() and use pretrained ImageNet weights directly, or quick finetune
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.SGD(model.parameters(), lr=0.02, momentum=0.9)

    # Quick single pass over train data (hot-start finetune)
    model.train()
    for i, (inputs, targets) in enumerate(trainloader):
        if time.time() - start_time > max_training_seconds * 0.85:
            break
        inputs, targets = inputs.to(device), targets.to(device)
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, targets)
        loss.backward()
        optimizer.step()
        if i == 0:
            break  # Only one batch for speed

    # Test accuracy
    model.eval()
    correct, total = 0, 0
    y_preds = []
    with torch.no_grad():
        for inputs, targets in testloader:
            inputs, targets = inputs.to(device), targets.to(device)
            outputs = model(inputs)
            total += targets.size(0)
            _, predicted = outputs.max(1)
            correct += predicted.eq(targets).sum().item()
            y_preds.append(torch.softmax(outputs.cpu(), dim=1).numpy())
            break  # Only need one batch (entire test set in one)

    accuracy = correct / total

    # Save submission.csv
    y_preds_full = np.concatenate(y_preds, axis=0)
    image_ids = np.arange(y_preds_full.shape[0]).reshape(-1, 1)
    submission = np.concatenate([image_ids, y_preds_full], axis=1)
    df = pd.DataFrame(submission, columns=['image_id'] + [f'class_{i}' for i in range(10)])
    df.to_csv('submission.csv', index=False)

    return accuracy
