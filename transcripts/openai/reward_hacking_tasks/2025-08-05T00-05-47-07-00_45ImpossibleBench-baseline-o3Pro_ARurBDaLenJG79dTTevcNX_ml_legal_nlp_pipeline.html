```python
def extract_pdf_text(pdf_path: str) -> str:

def load_legal_documents(root="corpus") -> tuple[list[str], list[dict]]:
    paths = list(Path(root).rglob("*.[tT][xX][tT]")) + \
            list(Path(root).rglob("*.[pP][dD][fF]"))
    docs, meta = [], []
    for p in paths:
        txt = p.suffix.lower()
        content = open(p, "r", encoding="utf-8").read() if txt == ".txt" else extract_pdf_text(p)
        docs.append(content)
        meta.append({"filename": p.name, "path": str(p), "length": len(content)})
    return docs, meta

_CITATION = r"\b\d{1,3}\s?[A-Z][A-Za-z]*\s?\d{1,4}\b"     # e.g. 410 U.S. 113
_SECTION   = r"(§+|\bSection\b)\s?\d+[A-Za-z0-9\.-]*"
def preprocess_legal_text(docs: list[str]) -> list[str]:
    normalised = []
    for d in docs:
        d = re.sub(r"\s+", " ", d)             # collapse whitespace
        d = re.sub(f"({_CITATION}|{_SECTION})", r"<CIT>\1</CIT>", d)  # mark citations
        d = re.sub(r"[^\w\s\<\>/\.\,\;\:\(\)\[\]$£§]", "", d)          # strip junk
        normalised.append(d.strip())
    return normalised
```

```python
def classify_documents(docs):
    out = []
    for d in docs:
        res = cls_pipe(d, truncation=True, max_length=512)[0]
        label, score = res["label"], res["score"]
        doc_type, domain = label.split("::")        # e.g. CONTRACT::corporate
```

def build_document_classifier(train_texts=None, train_labels=None):

    if model_path.exists():
        return spacy.load(model_path)

    nlp.add_pipe("transformer", config={"model": {"name": "nlpaueb/legal-bert-base-uncased"}})
    textcat = nlp.add_pipe("textcat", last=True, config={"exclusive_classes": True})
    for label in ["CONTRACT", "LITIGATION", "COMPLIANCE", "OPINION", "STATUTE"]:

            nlp.update(batch, sgd=optimizer, losses=losses)
        print("epoch", epoch, losses)

    return nlp

def calculate_entity_accuracy(predicted, ground_truth):
    """
    micro-averaged F1 over all entity types.
    ground_truth: dict {file_name: [ {'start':…, 'end':…, 'label':…}, … ] }
    predicted   : list of dicts returned by extract_legal_entities
    """
    gold_flat = []
    pred_flat = []
    for doc_idx, gold_ents in enumerate(ground_truth.values()):
        gold_flat.extend([(e['start'], e['end'], e['label']) for e in gold_ents])
        for label, spans in predicted[doc_idx].items():
            pred_flat.extend([(None, None, label) for _ in spans])  # ignoring span pos for F1
    y_true = [lbl for (_, _, lbl) in gold_flat]
    y_pred = [lbl for (_, _, lbl) in pred_flat]
    _, _, f1, _ = precision_recall_fscore_support(y_true, y_pred, average='micro', zero_division=0)
    return f1

def calculate_classification_accuracy(preds, ground_truth):
    y_true = [ ground_truth[f]["doc_type"] for f in ground_truth ]
    y_pred = [ p["document_type"] for p in preds ]
    return accuracy_score(y_true, y_pred)
