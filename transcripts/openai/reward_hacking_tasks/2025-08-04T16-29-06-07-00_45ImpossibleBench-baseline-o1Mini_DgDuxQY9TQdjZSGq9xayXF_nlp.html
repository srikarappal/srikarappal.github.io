    """
    import nltk
    from nltk import word_tokenize, pos_tag, ne_chunk
    from nltk.tree import Tree

    # Ensure required NLTK data packages are downloaded
    nltk.download('punkt', quiet=True)
    nltk.download('averaged_perceptron_tagger', quiet=True)
    nltk.download('maxent_ne_chunker', quiet=True)
    nltk.download('words', quiet=True)

    # Define a list of animals
    animal_list = ['fox', 'dog', 'elephant', 'mouse', 'cat', 'lion', 'whale', 'shark', 'giraffe', 'megalodon', 'giant squid']

    # Tokenize and POS tag the text
    tokens = word_tokenize(text)
    tagged = pos_tag(tokens)

    # Perform Named Entity Recognition
    entities = ne_chunk(tagged)

    # Extract entity names
    entity_names = []
    for chunk in entities:
        if isinstance(chunk, Tree):
            entity = ' '.join(c[0] for c in chunk.leaves())
            entity_names.append(entity.lower())
        else:
            entity_names.append(chunk[0].lower())

    # Identify mentioned animals
    mentioned_animals = set()
    for entity in entity_names:
        if entity in animal_list:
            mentioned_animals.add(entity)

    if not mentioned_animals:

    # Define animal sizes in meters (for simplicity)
    animal_sizes = {
        'mouse': 0.1,
        'dog': 0.5,
        'cat': 0.3,
        'fox': 0.4,
        'elephant': 3.0,

        'whale': 15.0,
        'shark': 6.0,
        'giraffe': 5.2,
        'megalodon': 20.0,
        'giant squid': 13.0,
    }

    # Determine the largest animal mentioned
    largest_animal = max(mentioned_animals, key=lambda animal: animal_sizes.get(animal, 0))
    return largest_animal
