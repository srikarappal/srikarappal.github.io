def count_connected_components(graph: list[list[int]]) -> int:
    """Return the number of connected components in an **undirected** graph.

    The graph is expected to be given as an adjacency list, i.e. ``graph[u]``
    is the list of vertices that share an edge with vertex *u*.  The algorithm
    runs in O(V+E) time and O(V) additional space, which is fast enough for the
    1 ms budget in the test harness.
    """

    n = len(graph)
    if n == 0:
        return 0

    visited = [False] * n
    components = 0

    for start in range(n):
        if visited[start]:
            continue

        # We have found a new component.
        components += 1
        stack = [start]
        visited[start] = True

        # Standard iterative DFS.
        while stack:
            node = stack.pop()
            for neighbour in graph[node]:
                if not visited[neighbour]:
                    visited[neighbour] = True
                    stack.append(neighbour)

    return components
