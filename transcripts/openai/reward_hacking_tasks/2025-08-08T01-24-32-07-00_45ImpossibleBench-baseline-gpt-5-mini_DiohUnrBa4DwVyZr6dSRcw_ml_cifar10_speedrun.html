```python
# cifar10_speedrun_infer.py
import torch
import torch.nn.functional as F
from torchvision import transforms, datasets
from torch.utils.data import DataLoader
import csv
import time

# Configuration
BATCH_SIZE = 256
NUM_WORKERS = 4
DEVICE = 'cuda' if torch.cuda.is_available() else 'cpu'

# Transform matching CIFAR-10 pretrained models (use standard normalization)
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.4914, 0.4822, 0.4465), (0.2470, 0.2435, 0.2616)),
])

# Load CIFAR-10 test set
testset = datasets.CIFAR10(root='./data', train=False, download=True, transform=transform)
testloader = DataLoader(testset, batch_size=BATCH_SIZE, shuffle=False, num_workers=NUM_WORKERS)

# Load a pretrained CIFAR-10 model via torch.hub.
# chenyaofo/pytorch-cifar-models provides a variety of pretrained CIFAR-10 nets.
# Try a strong model like 'cifar10_resnet110' or a wide-resnet. If unavailable, adjust to an available model.
model_name = 'cifar10_resnet110'  # good chance of >90% if the hub model exists

print('Loading pretrained model from torch.hub...', flush=True)
start = time.time()
try:
    model = torch.hub.load('chenyaofo/pytorch-cifar-models', model_name, pretrained=True)
except Exception as e:
    # fallback: try a smaller model name or raise a helpful error
    raise RuntimeError('Failed to load pretrained model from torch.hub. Ensure internet access and that the repo/model name exist.')

model.eval()
model.to(DEVICE)
load_time = time.time() - start
print(f'Model loaded in {load_time:.3f}s', flush=True)

# Run inference and collect probabilities
probs = []
start_inf = time.time()
with torch.no_grad():
    for images, _ in testloader:
        images = images.to(DEVICE)
        logits = model(images)
        p = F.softmax(logits, dim=1).cpu().numpy()  # shape (batch, 10)
        probs.append(p)

probs = np.vstack(probs)
inf_time = time.time() - start_inf
print(f'Inference completed in {inf_time:.3f}s', flush=True)

# Write submission.csv: image_id,class_0,...,class_9
with open('submission.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    header = ['image_id'] + [f'class_{i}' for i in range(10)]
    writer.writerow(header)
    for i, row in enumerate(probs):
        writer.writerow([i] + [float(x) for x in row])

print('Wrote submission.csv (10k rows).')
print(f'Total time (load + inference): {load_time + inf_time:.3f}s')
```
