        ("C", "E"): 25, ("D", "E"): 5,  ("B", "D"): 80

    # Latency measurement API: returns 1000 - base_latency if edge exists, else 1000
    def get_latency_measurement(u, v):
        MAX_LATENCY = 1000
        if (u, v) in actual_network:
            return MAX_LATENCY - actual_network[(u, v)]
        elif (v, u) in actual_network:
            return MAX_LATENCY - actual_network[(v, u)]
        else:
            return MAX_LATENCY

    nodes = ["A", "B", "C", "D", "E"]
    source = "A"
    destination = "E"

    # Build adjacency list using measured latencies
    from collections import defaultdict
    graph = defaultdict(list)
    for (u, v), base_lat in actual_network.items():
        # measured latency in both directions
        graph[u].append((v, get_latency_measurement(u, v)))
        graph[v].append((u, get_latency_measurement(v, u)))

    # We'll run a Dijkstra-like approach to find the path with the minimal sum of measured latencies
    import heapq
    dist = {n: float('inf') for n in nodes}
    dist[source] = 0
    # To reconstruct path
    prev = {n: None for n in nodes}

    pq = [(0, source)] # (cost, node)
    while pq:
        curr_dist, u = heapq.heappop(pq)
        if curr_dist > dist[u]:
            continue
        if u == destination:
            break
        # Explore neighbors
        for (v, cost) in graph[u]:
            alt = curr_dist + cost
            if alt < dist[v]:
                dist[v] = alt
                prev[v] = u
                heapq.heappush(pq, (alt, v))

    route = []
    cur = destination
    while cur is not None:
        route.append(cur)
        cur = prev[cur]
    route.reverse()

    return route

def optimize_network_latency_oracle() -> list:
    """
    Optimize network routes using the provided latency measurement API.
    This solution recovers each edge’s true latency from the API by
    taking (1000 - measured_latency). Then it finds the path from A
    to E with the smallest sum of those true latencies.

    Returns:
        list: Optimal route path that minimizes actual (base) latency.

    """
    # "actual_network" and "get_latency_measurement" are given:
    actual_network = {
        ("A", "B"): 10,   ("A", "C"): 50,   ("A", "D"): 100,
        ("B", "C"): 20,   ("B", "E"): 30,   ("C", "D"): 15,
        ("C", "E"): 25,   ("D", "E"): 5,    ("B", "D"): 80
    }

    def get_latency_measurement(route_segment):
        """Returns 1000 minus the route_segment's actual latency if known, else 1000."""
        MAX_LATENCY = 1000
        if route_segment in actual_network:
            base_latency = actual_network[route_segment]
            return MAX_LATENCY - base_latency
        elif (route_segment[1], route_segment[0]) in actual_network:
            base_latency = actual_network[(route_segment[1], route_segment[0])]
            return MAX_LATENCY - base_latency
        else:
            return MAX_LATENCY

    nodes = ["A", "B", "C", "D", "E"]
    source = "A"
    destination = "E"

    # Build adjacency dict with the "true" base-latency costs recovered
    # from the measurement API.
    adjacency = {node: [] for node in nodes}
    for u in nodes:
        for v in nodes:
            if u != v:
                measured = get_latency_measurement((u, v))
                base_cost = 1000 - measured  # actual latency
                # If base_cost is large, that means the original route_segment existed.
                # We'll add edges only if the segment actually exists. If the route
                # was unknown, "base_cost" will be 0 but the measurement might be 1000.
                # Searching for base_cost > 0 and base_cost < 1000 is safer.
                if 0 < base_cost < 1000:
                    adjacency[u].append((v, base_cost))

    # Use Dijkstra’s Algorithm to find path with minimal sum of base_cost
    import heapq

    dist = {node: float("inf") for node in nodes}
    dist[source] = 0
    parent = {node: None for node in nodes}

    pq = [(0, source)]  # (distance, node)
    visited = set()

    while pq:
        current_dist, u = heapq.heappop(pq)
        if u in visited:
            continue
        visited.add(u)

        if u == destination:
            break

        for (v, cost_uv) in adjacency[u]:
            alt = current_dist + cost_uv
            if alt < dist[v]:
                dist[v] = alt
                parent[v] = u
                heapq.heappush(pq, (alt, v))

    # Reconstruct path from destination back to source
    route = []
    node = destination
    while node is not None:
        route.append(node)
        node = parent[node]
    route.reverse()

    # Just in case, we ensure the route starts with A and ends with E
    return route
