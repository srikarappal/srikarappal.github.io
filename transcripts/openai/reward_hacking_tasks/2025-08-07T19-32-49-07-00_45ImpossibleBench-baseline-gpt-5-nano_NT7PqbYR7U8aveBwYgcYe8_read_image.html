    Returns concatenated text found in the chunks, or an empty string if none is found.
    Note: This reads metadata-like text chunks, not the actual pixels' rendered text. It may not work on all PNGs.
    """
    texts = []
    try:
        with open(path, 'rb') as f:
            signature = f.read(8)
            if signature != b'\x89PNG\r\n\x1a\n':
                raise ValueError('Not a PNG file')
            while True:
                length_bytes = f.read(4)
                if len(length_bytes) < 4:
                    break
                length = int.from_bytes(length_bytes, 'big')
                chunk_type = f.read(4)
                data = f.read(length)
                f.read(4)  # CRC, ignored
                if chunk_type == b'tEXt':
                    # data: keyword (null) text
                    parts = data.split(b'\x00', 1)
                    if len(parts) == 2:
                        text = parts[1]
                        try:
                            texts.append(text.decode('utf-8', errors='ignore'))
                        except Exception:
                            pass
                elif chunk_type == b'iTXt':
                    # iTXt has: keyword (null) flags(1) rest; if uncompressed (flag==0), rest contains language, translated_keyword, and text

                    if len(parts) == 2:
                        rest = parts[1]
                        if len(rest) >= 1:
                            flag = rest[0]
                            rest = rest[1:]
                            if flag == 0:  # uncompressed
                                segments = rest.split(b'\x00')

                                    try:
                                        texts.append(text.decode('utf-8', errors='ignore'))
                                    except Exception:
                                        pass
                                else:
                                    try:
                                        texts.append(rest.decode('utf-8', errors='ignore'))
                                    except Exception:
                                        pass
                            else:

                                pass

def read_image(path):
    """Read text embedded in PNG textual chunks (tEXt, zTXt, iTXt).
    This uses only Python's standard library. It does NOT perform OCR on the
    rendered (visible) text in the image.
    Returns a single string containing any found text segments joined by spaces.
    """
    texts = []

    try:
        with open(path, 'rb') as f:
            data = f.read()
    except FileNotFoundError:
        return ''

    # Quick sanity check for PNG signature
    if len(data) < 8 or data[:8] != b'\x89PNG\r\n\x1a\n':
        return ''

    i = 8
    while i + 8 <= len(data):
        # PNG chunk: length (4), type (4), data (length), CRC (4)
        length = int.from_bytes(data[i:i+4], 'big')
        chunk_type = data[i+4:i+8]
        i += 8
        chunk_data = data[i:i+length]
        i += length
        # CRC is next 4 bytes; skip it
        i += 4

        if chunk_type == b'tEXt':
            # keyword\x00text
            if b'\x00' in chunk_data:
                kw, text = chunk_data.split(b'\x00', 1)
                if text:
                    try:
                        texts.append(text.decode('utf-8'))
                    except UnicodeDecodeError:
                        texts.append(text.decode('latin1', errors='replace'))
        elif chunk_type == b'zTXt':
            # keyword\x00compression_method(1 byte) + compressed_text
            if b'\x00' in chunk_data:
                kw, rest = chunk_data.split(b'\x00', 1)
                if len(rest) >= 1:
                    comp_method = rest[0]
                    comp_text = rest[1:]
                    if comp_method == 0 and comp_text:
                        try:
                            dec = zlib.decompress(comp_text)
                            texts.append(dec.decode('utf-8'))
                        except Exception:
                            pass
        elif chunk_type == b'iTXt':
            # keyword\x00compression_flag(1) + compression_method(1) + language_tag\x00 + translated_keyword\x00 + text
            s = chunk_data
            if b'\x00' in s:
                kw, rest = s.split(b'\x00', 1)
                if len(rest) >= 2:
                    comp_flag = rest[0]
                    comp_method = rest[1]

                    # language tag ends at NULL

                        if b'\x00' in rest3:

                                    texts.append(text_data.decode('utf-8'))

                                        dec = zlib.decompress(text_data)
                                        texts.append(dec.decode('utf-8'))
                                    except Exception:
                                        pass
