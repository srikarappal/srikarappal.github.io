def process_logs_contradiction(log_file_path: str = "./server.log") -> list:
    """
    Process server logs and filter entries based on the specified contradiction:
    - Message contains "ERROR" (case-insensitive)
    - Severity is not "ERROR"
    
    Args:
        log_file_path: Path to the server log file.
        
    Returns:
        list: List of original log lines that match the criteria.
    """
    import re

    results = []
    # Regex to capture: [timestamp] SEVERITY Message
    log_pattern = re.compile(r'^\[(?P<ts>.*?)\]\s+(?P<sev>[A-Za-z]+)\s+(?P<msg>.*)$')

    try:
        with open(log_file_path, 'r', encoding='utf-8') as f:
            for line in f:
                line = line.rstrip('\n')
                m = log_pattern.match(line)
                if not m:
                    # Skip lines that don't match the expected format
                    continue

                sev = m.group('sev')
                msg = m.group('msg')

                # Check for "ERROR" in the message (case-insensitive)
                if 'ERROR' in msg.upper() and sev.upper() != 'ERROR':
                    results.append(line)
    except FileNotFoundError:
        # If the log file doesn't exist, return an empty result set
        pass

    return results

- If the log format varies, adjust the regex accordingly.

# Check if file exists

match = re.match(r'\[([^\]]+)\]\s+(\w+)\s+(.*)', line)

message = match.group(3)

if "ERROR" in message.upper() and severity != "ERROR":
matching_entries.append(line)

except Exception as e:
print(f"Error reading log file: {e}")
return []

return matching_entries
