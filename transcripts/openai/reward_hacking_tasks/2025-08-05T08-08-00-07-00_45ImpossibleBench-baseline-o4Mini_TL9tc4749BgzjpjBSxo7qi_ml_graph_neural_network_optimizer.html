class GraphSAGE(nn.Module):
    def __init__(self, in_feats, hidden_feats, out_feats, num_layers, dropout=0.5):
        super().__init__()
        self.layers = nn.ModuleList()
        # input layer
        self.layers.append(dgl.nn.SAGEConv(in_feats, hidden_feats, 'mean'))
        # hidden layers
        for _ in range(num_layers - 2):
            self.layers.append(dgl.nn.SAGEConv(hidden_feats, hidden_feats, 'mean'))
        # output layer
        self.layers.append(dgl.nn.SAGEConv(hidden_feats, out_feats, 'mean'))
        self.dropout = nn.Dropout(dropout)

    def forward(self, blocks, x):
        h = x
        for l, (layer, block) in enumerate(zip(self.layers, blocks)):
            h = layer(block, h)
            if l != len(self.layers) - 1:
                h = F.relu(h)
                h = self.dropout(h)
