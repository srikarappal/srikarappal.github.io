class DermoscopyDataset(Dataset):
    def __init__(self, df, img_dir, transforms=None):
        self.df, self.img_dir, self.transforms = df, Path(img_dir), transforms

    @staticmethod
    def dull_razor(img):
        gray = cv2.cvtColor(img, cv2.COLOR_RGB2GRAY)
        kernel = cv2.getStructuringElement(1, (17,17))
        blackhat = cv2.morphologyEx(gray, cv2.MORPH_BLACKHAT, kernel)
        _, mask = cv2.threshold(blackhat, 10, 255, cv2.THRESH_BINARY)
        mask = cv2.dilate(mask, None, iterations=1)
        img_nohair = cv2.inpaint(img, mask, 3, cv2.INPAINT_TELEA)
        return img_nohair

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img = cv2.imread(str(self.img_dir/row.image_id))
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        img = self.dull_razor(img)                          # ↖ hair removal
        if self.transforms:
            img = self.transforms(image=img)['image']
        label = torch.tensor(row.label, dtype=torch.long)
        return img, label

    def __len__(self):
        return len(self.df)

class SkinClassifier(nn.Module):
    def __init__(self, n_classes=2):
        super().__init__()
        self.backbone = timm.create_model('efficientnet_b0',
                                          pretrained=True,
                                          in_chans=3)
        n_features = self.backbone.classifier.in_features
        self.backbone.classifier = nn.Identity()
        self.head = nn.Sequential(
            nn.Dropout(0.3),
            nn.Linear(n_features, n_classes))
    def forward(self, x):
        x = self.backbone(x)
        return self.head(x)

def train_one_epoch(model, loader, optimizer, criterion, scaler):
    model.train(); running=0
    for x,y in loader:
        x,y=x.to(device), y.to(device)
        optimizer.zero_grad()
        with torch.cuda.amp.autocast():
            out = model(x); loss = criterion(out, y)
        scaler.scale(loss).backward()
        scaler.step(optimizer); scaler.update()
        running += loss.item()*x.size(0)
    return running/len(loader.dataset)

@torch.no_grad()
def validate(model, loader, criterion):
    model.eval(); preds, trues, loss_sum = [],[],0
    for x,y in loader:
        x,y=x.to(device), y.to(device)
        with torch.cuda.amp.autocast():
            out = model(x); loss = criterion(out, y)
        loss_sum += loss.item()*x.size(0)
        preds.extend(torch.argmax(out,1).cpu().numpy())
        trues.extend(y.cpu().numpy())
    acc = accuracy_score(trues, preds)
    sens = recall_score(trues, preds, pos_label=1)
    spec = recall_score(trues, preds, pos_label=0)
    return loss_sum/len(loader.dataset), acc, sens, spec

def train_folds(config):
    df = pd.read_csv('dataset/labels.csv')
    df['label'] = df['diagnosis'].map({'benign':0,'malignant':1})
    skf = StratifiedKFold(n_splits=config['folds'], shuffle=True,
                          random_state=42)

    metrics = []; fold=0
    for tr_idx, val_idx in skf.split(df.image_id, df.label):
        fold+=1
        tr_df, va_df = df.iloc[tr_idx], df.iloc[val_idx]

        tr_ds = DermoscopyDataset(tr_df, 'dataset/images', train_tfms)
        va_ds = DermoscopyDataset(va_df, 'dataset/images', valid_tfms)
        tr_dl = DataLoader(tr_ds, batch_size=config['bs'],
                           shuffle=True, num_workers=4)
        va_dl = DataLoader(va_ds, batch_size=config['bs']*2,
                           shuffle=False, num_workers=4)

        model = SkinClassifier().to(device)
        optimizer = torch.optim.AdamW(model.parameters(),
                                      lr=config['lr'], weight_decay=0.01)
        criterion = nn.CrossEntropyLoss()
        scaler = torch.cuda.amp.GradScaler()
        best_acc = 0

        for epoch in range(config['epochs']):
            tr_loss = train_one_epoch(model, tr_dl, optimizer, criterion, scaler)
            val_loss, acc, sens, spec = validate(model, va_dl, criterion)

            if acc>best_acc:
                best_acc=acc
                torch.save(model.state_dict(), f'effnet_fold{fold}.pt')
            if epoch%5==0:
                print(f'F{fold} Ep{epoch} | '
                      f'TL {tr_loss:.3f} VL {val_loss:.3f} '
                      f'Acc {acc:.3f} Sens {sens:.3f} Spec {spec:.3f}')

        metrics.append(best_acc)
    print('CV accuracy:', np.mean(metrics))

if __name__=='__main__':
    cfg = dict(folds=5, bs=32, lr=2e-4, epochs=25)
    train_folds(cfg)

def build_classifier_model(n_classes=2, meta_features=3):
    # Image branch
    img_in = Input(shape=(IMG_SIZE, IMG_SIZE, 3), name='img_in')
    base   = EfficientNetB5(weights='imagenet', include_top=False)(img_in)
    gap    = GlobalAveragePooling2D()(base)
    img_feat = BatchNormalization()(gap)

    # Metadata branch
    meta_in   = Input(shape=(meta_features,), name='meta_in')
    meta_dense = Dense(16, activation='relu')(meta_in)

    x = Concatenate()([img_feat, meta_dense])
    x = Dropout(0.4)(x)
    out = Dense(n_classes, activation='softmax')(x)

    model = Model(inputs=[img_in, meta_in], outputs=out)
    return model
