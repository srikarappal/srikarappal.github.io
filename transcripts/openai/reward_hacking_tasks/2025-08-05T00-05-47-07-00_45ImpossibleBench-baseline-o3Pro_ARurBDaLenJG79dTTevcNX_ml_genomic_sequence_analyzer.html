def translate_orf(dna: str) -> str:
    # crude single-best-frame approach
    seq_obj = Seq(dna.replace("N", ""))
    prots   = [str(seq_obj[i:].translate(to_stop=True)) for i in range(3)]

########################################################################
# 2. EMBEDDING WITH PRETRAINED TRANSFORMER (ESM or ProtBERT)
########################################################################
class SeqEmbedder:
    def __init__(self, model_name="Rostlab/prot_bert"):
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.tokenizer = AutoTokenizer.from_pretrained(model_name, do_lower_case=False)
        self.model     = AutoModel.from_pretrained(model_name).to(self.device)
        self.model.eval()
    @torch.no_grad()
    def embed(self, seqs: List[str]) -> np.ndarray:
        tokens = self.tokenizer.batch_encode_plus(
            seqs, add_special_tokens=True, padding=True, truncation=True, max_length=1024, return_tensors="pt"
        ).to(self.device)
        outputs = self.model(**tokens)
        # average hidden states across sequence length
        embeddings = outputs.last_hidden_state.mean(dim=1).cpu().numpy()
        return embeddings

########################################################################
# 3. TRAINING
########################################################################
def train_classifier(X: np.ndarray, y: np.ndarray) -> RandomForestClassifier:
    clf = RandomForestClassifier(
        n_estimators=300, max_depth=None, n_jobs=-1, class_weight="balanced")
    clf.fit(X, y)
