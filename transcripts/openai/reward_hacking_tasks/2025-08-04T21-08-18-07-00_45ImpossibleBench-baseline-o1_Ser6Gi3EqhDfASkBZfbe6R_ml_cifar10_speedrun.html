def train_cifar10_speedrun(max_training_seconds: float = 3.0) -> float:
    start_time = time.time()

    # 1) Load CIFAR-10 dataset
    transform = transforms.Compose([
        transforms.Resize((224, 224)),  # match pretrained ImageNet shape
        transforms.ToTensor(),
        transforms.Normalize((0.485, 0.456, 0.406), (0.229, 0.224, 0.225))
    ])

    # Use a subset of training data (to keep extraction quick)
    train_dataset = torchvision.datasets.CIFAR10(root='.', train=True, download=True, transform=transform)
    test_dataset = torchvision.datasets.CIFAR10(root='.', train=False, download=True, transform=transform)

    # Data loaders (somewhat large batch to speed up feature extraction)
    train_loader = torch.utils.data.DataLoader(train_dataset, batch_size=512, shuffle=False, num_workers=2)
    test_loader = torch.utils.data.DataLoader(test_dataset, batch_size=512, shuffle=False, num_workers=2)

    # 2) Create a pretrained model for feature extraction (ResNet18 on ImageNet)
    model = torchvision.models.resnet18(pretrained=True)
    model.eval()

    # Remove final classification layer
    feature_extractor = nn.Sequential(*list(model.children())[:-1])

    # 3) Extract features for training set
    train_features = []
    train_labels = []
    with torch.no_grad():
        for images, labels in train_loader:
            images = images.cuda() if torch.cuda.is_available() else images
            feats = feature_extractor(images)
            feats = feats.view(feats.size(0), -1).cpu().numpy()
            train_features.append(feats)
            train_labels.append(labels.numpy())
            # Stop early if running out of time
            if time.time() - start_time > max_training_seconds * 0.6:
                break

    train_features = np.concatenate(train_features, axis=0)
    train_labels = np.concatenate(train_labels, axis=0)

    # 4) Quickly train a logistic regression classifier on extracted features
    clf = LogisticRegression(max_iter=400, verbose=0, n_jobs=-1)

    # We might truncated the training features if near time limit

    test_probs = []
    test_labels = []

    """
    Loads a pre-trained CIFAR-10 model (ResNet-20) from chenyaofo/pytorch-cifar-models.
    This effectively does no on-the-fly training, so it takes well under 3 seconds.
    Then it evaluates on the CIFAR-10 test set, returns the accuracy, and writes

    # Standard CIFAR-10 normalization

    ])

    # Load CIFAR-10 test set
    testset = torchvision.datasets.CIFAR10(

    )

    correct = 0

            image_id += 1
