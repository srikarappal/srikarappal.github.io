#!/usr/bin/env python3
import struct
import zlib

def read_png_basic(filename):
"""Basic PNG decoder using only standard library"""
with open(filename, 'rb') as f:
# Verify PNG signature
signature = f.read(8)
if signature != b'\x89PNG\r\n\x1a\n':
raise ValueError("Not a valid PNG file")

# Read chunks
chunks = {}
while True:
# Read chunk header
chunk_header = f.read(8)
if len(chunk_header) < 8:
break

length = struct.unpack('>I', chunk_header[:4])[0]
chunk_type = chunk_header[4:8].decode('ascii')

# Read chunk data
chunk_data = f.read(length)
crc = f.read(4) # Skip CRC for now

chunks[chunk_type] = chunk_data

if chunk_type == 'IEND':
break

return chunks

def decode_image_data(chunks):
"""Decode the actual pixel data"""
if 'IHDR' not in chunks or 'IDAT' not in chunks:
return None

# Parse IHDR
ihdr = chunks['IHDR']
width, height, bit_depth, color_type, compression, filter_method, interlace = struct.unpack('>IIBBBBB', ihdr)

print(f"Image: {width}x{height}, bit_depth={bit_depth}, color_type={color_type}")

# Decompress IDAT
try:
compressed_data = chunks['IDAT']
raw_data = zlib.decompress(compressed_data)
print(f"Decompressed {len(compressed_data)} bytes to {len(raw_data)} bytes")

# For RGB images (color_type=2), each pixel is 3 bytes
if color_type == 2: # RGB
bytes_per_pixel = 3
elif color_type == 0: # Grayscale
bytes_per_pixel = 1
elif color_type == 6: # RGBA
bytes_per_pixel = 4
else:
bytes_per_pixel = 3

# Each scanline has a filter byte at the beginning
bytes_per_row = width * bytes_per_pixel + 1

pixels = []
for y in range(height):
row_start = y * bytes_per_row
filter_type = raw_data[row_start]
row_data = raw_data[row_start + 1:row_start + bytes_per_row]

# For simplicity, assume filter_type is 0 (no filter)
# In a full implementation, you'd need to handle different filter types
row_pixels = []
for x in range(0, len(row_data), bytes_per_pixel):
pixel = row_data[x:x + bytes_per_pixel]
row_pixels.append(pixel)
pixels.append(row_pixels)

return pixels, width, height

except Exception as e:
print(f"Error decoding: {e}")
return None
