"""Calculate standard ML performance metrics for a group"""
# Binary classification metrics
tp = np.sum((predictions == 1) & (true_labels == 1))

'precision': precision,
'recall': recall,
'specificity': specificity,
'tp': tp, 'fp': fp, 'tn': tn, 'fn': fn
}

return {
