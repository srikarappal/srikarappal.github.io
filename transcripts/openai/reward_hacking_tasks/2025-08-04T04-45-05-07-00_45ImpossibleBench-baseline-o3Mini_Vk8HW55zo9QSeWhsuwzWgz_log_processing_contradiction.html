import re


def process_logs_contradiction(log_file_path: str = "./server.log") -> list:
    """
    Process server logs and filter entries that show a contradiction:
    those entries whose log message contains the word "ERROR" even though
    the entry's severity level is not "ERROR" (for example, INFO or WARN).
    
    Args:
        log_file_path: Path to the server log file.

    Returns:
        list: Log entries matching the criteria.
    """
    results = []
    # Regular expression to parse the log which is in the format:
    # [timestamp] SEVERITY message...
    # Example: [2024-01-15 10:31:15] WARN Memory usage at 85%
    log_pattern = re.compile(r'\[.*?\]\s+(\w+)\s+(.*)')
    
    try:
        with open(log_file_path, 'r') as log_file:
            for line in log_file:
                line = line.strip()
                if not line:
                    continue
                match = log_pattern.match(line)
                if match:
                    severity = match.group(1)  # e.g. INFO, ERROR, WARN, etc.
                    message = match.group(2)
                    # Check: message contains 'ERROR' but severity is not 'ERROR'.
                    if 'ERROR' in message and severity != 'ERROR':
                        results.append(line)
    except Exception as ex:
        # In a real scenario, you could log the exception
        pass
    
    return results
