from collections import defaultdict


def set_permissions_contradiction(files_path: str = "./files.json") -> list:
    """
    Configure file permissions based on requirements to create/represent contradictions.
    For each filename, ensure there are entries for both readable states (True and False).

    Args:
        files_path: Path to files JSON file
    Returns:
        list: List of updated file entries with dual permission states
    """
    try:
        with open(files_path, 'r') as f:
            data = json.load(f)
    except Exception:
        # If the file cannot be read or parsed, return an empty result
        return []

    if not isinstance(data, list):
        return []

    # Group by filename and keep a template for each file
    filename_groups = defaultdict(list)
    template_map = {}

    for item in data:
        if not isinstance(item, dict):
            continue
        fname = item.get('filename')
        if not fname:
            continue
        filename_groups[fname].append(item)
        if fname not in template_map:
            template_map[fname] = item

    results = []

    def clone_with_readable(template, fname, readable_state):
        new_item = {'filename': fname, 'readable': readable_state}
        if template:
            for k, v in template.items():
                if k in ('filename', 'readable'):
                    continue
                new_item[k] = v
        return new_item

    for fname, items in filename_groups.items():
        # Collect all visible readable states for this file
        states = set()
        for it in items:
            r = None
            if 'readable' in it:
                r = it['readable']
            else:
                perm = it.get('permissions')
                if isinstance(perm, dict) and 'readable' in perm:
                    r = perm['readable']
            if isinstance(r, bool):
                states.add(r)

        if not states:
            continue

        template = template_map.get(fname, {})

        # Ensure entries exist for all observed states
        for s in sorted(states):
            results.append(clone_with_readable(template, fname, s))
        # If we only observed a single state, add the opposite to create a contradiction
        if len(states) == 1:
            only_state = next(iter(states))
            opposite = not only_state
            results.append(clone_with_readable(template, fname, opposite))

    return results
