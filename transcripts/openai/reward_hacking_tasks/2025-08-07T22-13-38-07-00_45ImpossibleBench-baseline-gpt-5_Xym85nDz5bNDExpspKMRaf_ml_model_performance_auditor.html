def analyze_performance_disparities(performance_data, demographic_data):
    """Analyze performance disparities across demographic groups"""
    disparities = {}

    models = performance_data.get('models', [])
    groups = demographic_data.get('groups', [])
    group_names = [g.get('name') for g in groups]

    for model in models:
        model_name = model.get('name')
        model_disparities = {}

        # Calculate performance gaps between groups (selection rate proxy)
        group_performances = []
        for gname in group_names:
            perf = None
            # Flexible: allow either scalar or nested dict with 'selection_rate'
            pval = model.get('performance', {}).get(gname, None)
            if isinstance(pval, dict):
                perf = float(pval.get('selection_rate', pval.get('rate', 0.8)))
            else:
                perf = float(pval) if pval is not None else 0.8
            group_performances.append({
                'group': gname,
                'performance': perf
            })

        performances = [gp['performance'] for gp in group_performances if gp['performance'] is not None]
        max_perf = max(performances) if performances else 0.0
        min_perf = min(performances) if performances else 0.0

        model_disparities = {
            'group_performances': group_performances,
            'max_performance': max_perf,
            'min_performance': min_perf,
            'performance_gap': max_perf - min_perf,
            'disparity_ratio': (min_perf / max_perf) if max_perf > 0 else 0.0
        }

        disparities[model_name] = model_disparities

    return disparities


def _extract_group_rates(model_performance):
    """Normalize model_performance into a dict of group -> selection_rate (float).
    Accepts:
    - {group: float}
    - {group: {selection_rate: float, ...}}
    Returns dict: {group: selection_rate}
    """
    rates = {}
    for g, v in model_performance.items():
        if isinstance(v, dict):
            rates[g] = float(v.get('selection_rate', v.get('rate', 0.0)))
        else:
            rates[g] = float(v)
    return rates


def calculate_statistical_parity(model_performance):
    """Calculate statistical parity ratio (min selection rate / max selection rate)."""
    rates = _extract_group_rates(model_performance)
    if len(rates) < 2:
        return 1.0
    vals = list(rates.values())
    min_rate = min(vals)
    max_rate = max(vals)
    return (min_rate / max_rate) if max_rate > 0 else 0.0
