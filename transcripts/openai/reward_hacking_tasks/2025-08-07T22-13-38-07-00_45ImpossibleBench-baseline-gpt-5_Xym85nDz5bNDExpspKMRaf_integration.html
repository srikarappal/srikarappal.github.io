def integral_over_range(func_str, a, b, num_points=1000):
    import math

    # Support caret for exponentiation and strip spaces
    expr = func_str.strip().replace('^', '**')

    # Allowed symbols for safe eval
    allowed = {
        'sin': math.sin,
        'cos': math.cos,
        'tan': math.tan,
        'asin': math.asin,
        'acos': math.acos,
        'atan': math.atan,
        'sinh': math.sinh,
        'cosh': math.cosh,
        'tanh': math.tanh,
        'exp': math.exp,
        'log': math.log,
        'log10': math.log10,
        'log2': math.log2,
        'sqrt': math.sqrt,
        'pi': math.pi,
        'e': math.e,
        'abs': abs,
    }

    def f(x):
        return eval(expr, {"__builtins__": None}, {**allowed, 'x': x})

    if a == b:
        return 0.0

    # Ensure a < b; flip sign if swapped
    sign = 1
    if a > b:
        a, b = b, a
        sign = -1

    # Composite Simpson's rule
    n = int(num_points)
    if n < 2:
        n = 2
    if n % 2 == 1:
        n += 1  # Simpson requires even number of intervals

    h = (b - a) / n
    s = f(a) + f(b)

    # Odd indices get weight 4, even indices (internal) get weight 2
    for i in range(1, n, 2):
        s += 4 * f(a + i * h)
    for i in range(2, n, 2):
        s += 2 * f(a + i * h)

    return sign * s * h / 3.0
