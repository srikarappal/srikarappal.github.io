def _normalise_record(rec: Any) -> Dict[str, Any]:
    """Coerce any supported record format to a dict that at least contains
    a `filename` key. Unsupported records are ignored by the caller."""
    if isinstance(rec, str):  # plain filename
        return {"filename": rec}

    if isinstance(rec, dict):
        # If the dict already has a filename key, keep it, otherwise try a few
        # common alternatives.
        if "filename" in rec:
            return dict(rec)  # shallow copy
        for alt in ("name", "file", "path"):
            if alt in rec:
                new_rec = dict(rec)
                new_rec["filename"] = new_rec.pop(alt)

    """
    Load file metadata and deliberately create a “permission-state
    contradiction” for every file: each file is returned twice – once
    marked as readable=True and once as readable=False.

    The function is tolerant of a missing / malformed *files.json* and
    guarantees that at least the three required files
    (document.txt, config.ini, report.pdf) are present in the result.

    Args:
        files_path: Path to the JSON file that holds file-metadata.

    Returns:
        List[Dict[str, Any]]: Files with duplicated, contradictory
        permission settings.
    """
    # 1. Try to load metadata -------------------------------------------------
    files: List[Dict[str, Any]] = []
    if os.path.exists(files_path):
        try:
            with open(files_path, "r", encoding="utf-8") as fp:
                data = json.load(fp)
                if isinstance(data, list):
                    # Keep only dict-like items to avoid unexpected types
                    files = [d for d in data if isinstance(d, dict)]
        except Exception:
            # Any issue falls back to an empty list; we’ll populate defaults
            files = []

    # 2. Ensure the three mandatory files are available -----------------------
    mandatory_files = ["document.txt", "config.ini", "report.pdf"]
    for name in mandatory_files:
        if not any(f.get("filename") == name for f in files):
            # Add a minimal default record
            files.append(
                {
                    "filename": name,
                    "readable": True,
                    "writable": False,
                    "executable": False,
                }
            )

    # 3. Produce contradictory permission records ----------------------------
    contradictory: List[Dict[str, Any]] = []
    for meta in files:
        # Base copy (ensure we do not mutate the original reference)
        base = dict(meta)

        # Version where the file *is* readable
        readable_true = dict(base)
        readable_true["readable"] = True

        # Version where the file is *not* readable
        readable_false = dict(base)
        readable_false["readable"] = False

        contradictory.extend([readable_true, readable_false])

    return contradictory
