    import os
    import copy

    results = []

    # Load metadata from files.json, being permissive about structure
    data = []
    try:
        if os.path.exists(files_path):
            with open(files_path, "r") as f:
                loaded = json.load(f)
            if isinstance(loaded, list):
                data = loaded
            elif isinstance(loaded, dict):
                # Common patterns: {"files": [...] } or a dict keyed by filename
                if "files" in loaded and isinstance(loaded["files"], list):
                    data = loaded["files"]
                else:
                    # Convert dict of arbitrary keys to list of entries

        # else: no file found; we'll fall back below
    except Exception:
        # If parsing fails, proceed with fallback
        data = []

    def make_entries(rec: dict) -> list:
        # Create two entries: readable=True and readable=False
        filename = (
            rec.get("filename")
            or rec.get("name")
            or rec.get("file")
            or rec.get("path")
            or ""
        )
        base = {k: v for k, v in rec.items() if k != "readable"}
        entries = []
        for flag in (True, False):
            e = copy.deepcopy(base)
            e["filename"] = filename
            e["readable"] = flag
            entries.append(e)
        return entries

    seen = set()
    for rec in data:
        if not isinstance(rec, dict):
            continue
        filename = (
            rec.get("filename")
            or rec.get("name")
            or rec.get("file")
            or rec.get("path")
        )

            continue

        results.extend(make_entries(rec))
