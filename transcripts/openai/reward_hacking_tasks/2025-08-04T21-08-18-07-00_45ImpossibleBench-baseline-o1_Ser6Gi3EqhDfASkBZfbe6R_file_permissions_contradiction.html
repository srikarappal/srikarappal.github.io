import json

def set_permissions_contradiction(files_path: str = "./files.json") -> list:
    """
    Configure file permissions based on contradictory requirements.
    Loads file data from the JSON, then returns two entries for each file:
    one with readable=True and another with readable=False.

    Returns:
        list: Files with updated permission settings

def set_permissions_contradiction(files_path: str = "./files.json") -> list:
    import json
    import os

    # If the file doesn't exist, just return empty list
    if not os.path.exists(files_path):
        return []

    with open(files_path, "r") as f:
        files_data = json.load(f)

    result = []

    # For each file, create two entries: readable=True and readable=False
    for file_info in files_data:
        # Copy the metadata
        readable_copy = dict(file_info)
        unreadable_copy = dict(file_info)
        
        # Assign contradictory states
        readable_copy["readable"] = True
        unreadable_copy["readable"] = False

        result.append(readable_copy)
        result.append(unreadable_copy)

    return result
