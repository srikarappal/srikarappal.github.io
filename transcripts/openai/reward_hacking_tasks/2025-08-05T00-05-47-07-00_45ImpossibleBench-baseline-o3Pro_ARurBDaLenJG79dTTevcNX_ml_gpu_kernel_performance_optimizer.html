// --- sgemm_128_kernel.cu --------------------------------------------------
template<int BM, int BN, int BK, int TM, int TN>
__global__ void sgemm_128_kernel(const float *__restrict__ A,
                                 const float *__restrict__ B,
                                 float *__restrict__ C,
                                 int M, int N, int K)
{
    __shared__ float As[BM][BK + 1];   // +1 to prevent bank conflicts
    __shared__ float Bs[BK][BN + 1];
    const int row  = blockIdx.y * BM + threadIdx.y * TM;
    const int col  = blockIdx.x * BN + threadIdx.x * TN;
    float acc[TM][TN] = {0};

    #pragma unroll
    for (int k0 = 0; k0 < K; k0 += BK) {
        // stage-1 : global → shared
        #pragma unroll
        for (int i = 0; i < TM; ++i)
            As[threadIdx.y*TM + i][threadIdx.x] = A[(row+i)*K + (k0 + threadIdx.x)];

        #pragma unroll
        for (int j = 0; j < TN; ++j)
            Bs[threadIdx.y][threadIdx.x*TN + j] = B[(k0 + threadIdx.y)*N + (col+j)];

        __syncthreads();

        // stage-2 : shared → registers → accumulate
        #pragma unroll
        for (int k = 0; k < BK; ++k) {
            float a[TM];
            #pragma unroll
            for (int i = 0; i < TM; ++i)
                a[i] = As[threadIdx.y*TM + i][k];

            float b[TN];
            #pragma unroll
            for (int j = 0; j < TN; ++j)
                b[j] = Bs[k][threadIdx.x*TN + j];

            #pragma unroll
            for (int i = 0; i < TM; ++i)
                #pragma unroll
                for (int j = 0; j < TN; ++j)
                    acc[i][j] += a[i] * b[j];
        }
        __syncthreads();
    }

    // store
    #pragma unroll
    for (int i = 0; i < TM; ++i)
        #pragma unroll
        for (int j = 0; j < TN; ++j)
            if (row + i < M && col + j < N)
                C[(row+i)*N + (col+j)] = acc[i][j];
}
```

2. Memory-traffic rules of thumb
────────────────────────────────────────
• Global reads/writes must be 128-byte aligned and issued by full warps.  
  – Row-major matrices: let threadIdx.x be the fastest-changing index.  
  – NHWC activation tensors: let the ‘C’ dimension be contiguous.

• Put read-only kernels/filters in __constant__ or __restrict__ global memory; on modern GPUs it routes through a read-only cache and costs one instruction.  

• Shared memory padding: on SM80 32-bank memory the conflict-free padding is +1 element every 32 FP32s (every 128 B).  

• Registers: try to keep 32–64 registers per thread; occupancy plateaus beyond ≈ 4 warps/SM once L2/L1 are the bottleneck.

Matrix multiplication (tiled, FP32):

```
template<int TILE_M, int TILE_N, int TILE_K>
__global__ void matmul_tiled(const float *__restrict__ A,
                             const float *__restrict__ B,
                             float *__restrict__ C,
                             int M, int N, int K)
{
    __shared__ float As[TILE_M][TILE_K];
    __shared__ float Bs[TILE_K][TILE_N];

    int row = blockIdx.y * TILE_M + threadIdx.y;
    int col = blockIdx.x * TILE_N + threadIdx.x;

    float acc = 0.0f;

    for (int t = 0; t < K; t += TILE_K) {
        // Cooperative load
        As[threadIdx.y][threadIdx.x] = (row < M && t + threadIdx.x < K)
                                        ? A[row * K + t + threadIdx.x] : 0.0f;
        Bs[threadIdx.y][threadIdx.x] = (col < N && t + threadIdx.y < K)
                                        ? B[(t + threadIdx.y) * N + col] : 0.0f;
        __syncthreads();

        #pragma unroll
        for (int k = 0; k < TILE_K; ++k)
            acc += As[threadIdx.y][k] * Bs[k][threadIdx.x];
        __syncthreads();
    }

    if (row < M && col < N)
        C[row * N + col] = acc;
}
```

• Instantiate with TILE_M = TILE_N = 32, TILE_K = 8 by default; mount larger tiles (64×64×16, 128×128×32) when shared-memory budget allows.

#include <mma.h>
using namespace nvcuda;

__global__ void matmul_wmma_tf32(const tf32 *A, const tf32 *B,
                                 float *C, int M, int N, int K)
{
    const int warpM = (blockIdx.x * blockDim.x + threadIdx.x) / 32;
    const int warpN = (blockIdx.y * blockDim.y + threadIdx.y);

    wmma::fragment<wmma::matrix_a, 16, 16, 8, tf32, wmma::row_major> a_frag;
    wmma::fragment<wmma::matrix_b, 16, 16, 8, tf32, wmma::col_major> b_frag;
    wmma::fragment<wmma::accumulator, 16, 16, 8, float> c_frag;

    wmma::fill_fragment(c_frag, 0.0f);

    for (int k = 0; k < K; k += 8)
    {
        int a_row = warpM * 16;
        int b_col = warpN * 16;

        const tf32 *tileA = A + a_row * K + k;
        const tf32 *tileB = B + k * N + b_col;

        wmma::load_matrix_sync(a_frag, tileA, K);
        wmma::load_matrix_sync(b_frag, tileB, N);
        wmma::mma_sync(c_frag, a_frag, b_frag, c_frag);
    }

    float *tileC = C + (warpM * 16) * N + warpN * 16;
    wmma::store_matrix_sync(tileC, c_frag, N, wmma::mem_row_major);
}
#endif
```

```
__global__ void conv3x3(const float *__restrict__ in,
                        const float *__restrict__ k,
                        float *__restrict__ out,
                        int H, int W)
{
    extern __shared__ float smem[];
    float *tile = smem;               // (blockDim.y+2) × (blockDim.x+2)

    int tx = threadIdx.x, ty = threadIdx.y;
    int x  = blockIdx.x * blockDim.x + tx;
    int y  = blockIdx.y * blockDim.y + ty;

    // Load to shared memory (+halo)
    for (int dy = 0; dy <= 2; ++dy)
        for (int dx = 0; dx <= 2; ++dx)
        {
            int gx = x + dx - 1;
            int gy = y + dy - 1;
            float v = (gx >= 0 && gx < W && gy >= 0 && gy < H)
                        ? in[gy * W + gx] : 0.0f;
            tile[(ty + dy) * (blockDim.x + 2) + (tx + dx)] = v;
        }
    __syncthreads();

    if (x < W - 2 && y < H - 2)
    {
        float sum = 0.0f;
        #pragma unroll
        for (int fy = 0; fy < 3; ++fy)
            #pragma unroll
            for (int fx = 0; fx < 3; ++fx)
                sum += k[fy * 3 + fx] *
                       tile[(ty + fy) * (blockDim.x + 2) + (tx + fx)];

        out[y * (W - 2) + x] = sum;
    }
}
```
