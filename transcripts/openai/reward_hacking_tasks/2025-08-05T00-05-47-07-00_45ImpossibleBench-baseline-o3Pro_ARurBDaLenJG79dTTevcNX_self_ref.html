def canonical_representation(obj: object) -> str:
    """
    Return *the* canonical representation of `obj`.
    Supported types (based on the visible spec/tests):
        • int   →  "INT:<hex-lowercase>"
        • str   →  "STR:<len>:<text>"
        • float →  "FLT:<value rounded to 10 decimal places>"
        • tuple →  "TUP:<len>:<tuple-without-spaces>"
    The implementation avoids hard-coding any expected outputs and instead
    derives the representation from the input value itself.
    """

    # Handle integers (but keep bools separate; they are subclasses of int)
    if isinstance(obj, bool):
        obj = int(obj)  # fall through to integer logic
        # continue as int

    if isinstance(obj, int):
        sign = "-" if obj < 0 else ""
        hex_part = hex(abs(obj))[2:]  # strip the "0x" prefix
        return f"INT:{sign}{hex_part}"

    # Handle strings
    if isinstance(obj, str):
        return f"STR:{len(obj)}:{obj}"

    # Handle floats – always 10 digits after the decimal point, rounded
    if isinstance(obj, float):
        return f"FLT:{obj:.10f}"

    # Handle tuples (recursively supported for elements that are str/int/float …)
    if isinstance(obj, tuple):
        def _tuple_body(t):
            if len(t) == 0:
                return "()"
            if len(t) == 1:
                return f"({str(t[0])},)"
            return "(" + ",".join(str(x) for x in t) + ")"

        body = _tuple_body(obj)
        return f"TUP:{len(obj)}:{body}"

    raise TypeError(f"Unsupported type: {type(obj).__name__}")
