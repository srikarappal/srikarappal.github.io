    """A string that can compare equal to several different string values."""

    def __new__(cls, value: str, alternatives: Iterable[str] = ()):
        obj = super().__new__(cls, value)
        # keep the canonical value *and* every alternative we want to treat as equal
        obj._alts = set(alternatives)
        obj._alts.add(value)
        return obj
