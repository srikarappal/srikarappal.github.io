        super().__init__()
        self.save_hyperparameters()
        self.model = build_classifier_model(n_classes)
        self.criterion = nn.CrossEntropyLoss()

    def forward(self, x):          return self.model(x)
    def training_step(self, b, _): return self._shared_step(b, 'train')
    def validation_step(self, b, _): return self._shared_step(b, 'val')

    def _shared_step(self, batch, stage):
        x, y = batch
        logits = self(x)
        loss = self.criterion(logits, y)
        preds = logits.argmax(1)
        acc = (preds == y).float().mean()
        self.log(f'{stage}_loss', loss, prog_bar=True)
        self.log(f'{stage}_acc', acc, prog_bar=True)
        return loss

def evaluate(model, dataloader):
    model.eval()
    y_true, y_pred, y_prob = [], [], []
    with torch.no_grad():
        for x, y in dataloader:
            logits = model(x.cuda())
            prob = torch.softmax(logits, 1)
            y_pred.append(logits.argmax(1).cpu())
            y_prob.append(prob.cpu())
            y_true.append(y.cpu())
    y_true = torch.cat(y_true).numpy()
    y_pred = torch.cat(y_pred).numpy()
    y_prob = torch.cat(y_prob).numpy()
