crypto_data['sma_20'] = crypto_data['close'].rolling(window=20).mean()
crypto_data['ema_12'] = crypto_data['close'].ewm(span=12).mean()
crypto_data['ema_26'] = crypto_data['close'].ewm(span=26).mean()

# RSI calculation
delta = crypto_data['close'].diff()
gain = (delta.where(delta > 0, 0)).rolling(window=14).mean()
loss = (-delta.where(delta < 0, 0)).rolling(window=14).mean()
rs = gain / loss
crypto_data['rsi'] = 100 - (100 / (1 + rs))

# MACD
crypto_data['macd'] = crypto_data['ema_12'] - crypto_data['ema_26']
crypto_data['macd_signal'] = crypto_data['macd'].ewm(span=9).mean()

# Bollinger Bands
crypto_data['bb_middle'] = crypto_data['close'].rolling(window=20).mean()
bb_std = crypto_data['close'].rolling(window=20).std()
crypto_data['bb_upper'] = crypto_data['bb_middle'] + (bb_std * 2)
crypto_data['bb_lower'] = crypto_data['bb_middle'] - (bb_std * 2)

# Volume indicators
crypto_data['volume_sma'] = crypto_data['volume'].rolling(window=20).mean()

# Add news sentiment
news_df = pd.DataFrame(news_data)
news_df['timestamp'] = pd.to_datetime(news_df['timestamp'])

# Aggregate sentiment by hour
news_df['hour'] = news_df['timestamp'].dt.floor('H')
hourly_sentiment = news_df.groupby('hour')['sentiment'].mean().reset_index()
hourly_sentiment.columns = ['timestamp', 'sentiment_score']

# Merge with crypto data
crypto_data = crypto_data.merge(hourly_sentiment, on='timestamp', how='left')
crypto_data['sentiment_score'] = crypto_data['sentiment_score'].fillna(0)
