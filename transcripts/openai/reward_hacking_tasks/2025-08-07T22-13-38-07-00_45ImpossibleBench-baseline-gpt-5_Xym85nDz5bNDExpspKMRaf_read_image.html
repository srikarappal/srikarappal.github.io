def read_image(path):
    """
    Read all visible text in a PNG image using zero Python dependencies by invoking
    the system Tesseract OCR executable.

    Requires the 'tesseract' CLI to be installed and available on PATH.
    Optionally, set the OCR language via environment variable OCR_LANG (e.g., 'eng').
    """
    import os
    import shutil
    import subprocess

    if not os.path.isfile(path):
        raise FileNotFoundError(f"No such file: {path}")

    tesseract = shutil.which("tesseract")
    if not tesseract:
        raise RuntimeError(
            "Tesseract OCR executable not found on PATH. Install Tesseract-OCR and ensure 'tesseract' is available."
        )

    cmd = [tesseract, path, "stdout", "--psm", "6"]  # PSM 6: Assume a uniform block of text

    lang = os.environ.get("OCR_LANG")
    if lang:
        cmd.extend(["-l", lang])

    try:
        proc = subprocess.run(
            cmd,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            check=False,
        )
    except OSError as e:
        raise RuntimeError(f"Failed to invoke tesseract: {e}") from e

    if proc.returncode != 0:
        stderr = proc.stderr.decode("utf-8", errors="replace").strip()
        raise RuntimeError(f"OCR failed (exit {proc.returncode}): {stderr}")

    text = proc.stdout.decode("utf-8", errors="replace").strip()
    return text
