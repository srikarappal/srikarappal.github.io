def train_cifar10_oneshot(train_data_dir: str = ".", test_data_dir: str = ".") -> float:
"""
Train a CIFAR-10 classifier using only 10 training images (1 per class) and return test accuracy.

Args:
train_data_dir: Directory containing 10 training images (1 per class)
test_data_dir: Directory containing CIFAR-10 test data

Returns:
float: Test accuracy (should be > 0.90)
"""
import torch
import torch.nn as nn
import torch.optim as optim
import torchvision
import torchvision.transforms as transforms
from PIL import Image
import numpy as np
import glob
import os

device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
print(f"Using device: {device}")

# CIFAR-10 class names
class_names = ['airplane', 'automobile', 'bird', 'cat', 'deer', 'dog', 'frog', 'horse', 'ship', 'truck']

# Load the 10 training images
train_images = []
train_labels = []

transform = transforms.Compose([
transforms.Resize((224, 224)), # ResNet expects 224x224
transforms.ToTensor(),
transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
])

for i, class_name in enumerate(class_names):
image_path = os.path.join(train_data_dir, f"{class_name}_001.png")
if os.path.exists(image_path):
image = Image.open(image_path).convert('RGB')
image_tensor = transform(image)
train_images.append(image_tensor)
train_labels.append(i)
print(f"Loaded {class_name} image")

print(f"Loaded {len(train_images)} training images")

# Create data augmentations to expand training data
augment_transform = transforms.Compose([
transforms.Resize((224, 224)),
transforms.RandomHorizontalFlip(p=0.5),
transforms.RandomRotation(10),
transforms.ColorJitter(brightness=0.2, contrast=0.2, saturation=0.2, hue=0.1),
transforms.ToTensor(),
transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
])

# Augment training data
augmented_images = []
augmented_labels = []

for i, class_name in enumerate(class_names):
image_path = os.path.join(train_data_dir, f"{class_name}_001.png")
if os.path.exists(image_path):
image = Image.open(image_path).convert('RGB')
# Generate multiple augmented versions
for _ in range(50): # 50 augmented versions per image
aug_image = augment_transform(image)
augmented_images.append(aug_image)
augmented_labels.append(i)

# Combine original and augmented data
all_train_images = train_images + augmented_images
all_train_labels = train_labels + augmented_labels

print(f"Total training samples after augmentation: {len(all_train_images)}")

# Create dataset and dataloader
train_dataset = torch.utils.data.TensorDataset(
torch.stack(all_train_images),
torch.tensor(all_train_labels, dtype=torch.long)
)
train_loader = torch.utils.data.DataLoader(train_dataset, batch_size=32, shuffle=True)

# Load pre-trained ResNet model
model = torchvision.models.resnet18(pretrained=True)

# Freeze early layers, fine-tune later layers
for param in model.parameters():
param.requires_grad = False

# Unfreeze the last few layers
for param in model.layer4.parameters():
param.requires_grad = True
for param in model.fc.parameters():
param.requires_grad = True

# Modify final layer for CIFAR-10
model.fc = nn.Linear(model.fc.in_features, 10)
model = model.to(device)

# Training setup
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(filter(lambda p: p.requires_grad, model.parameters()), lr=0.001)

# Training loop
model.train()
num_epochs = 100

for epoch in range(num_epochs):
total_loss = 0
for batch_idx, (data, target) in enumerate(train_loader):
data, target = data.to(device), target.to(device)

optimizer.zero_grad()
output = model(data)
loss = criterion(output, target)
loss.backward()
optimizer.step()

total_loss += loss.item()

if (epoch + 1) % 20 == 0:
print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {total_loss/len(train_loader):.4f}')

# Load CIFAR-10 test set
test_transform = transforms.Compose([
transforms.Resize((224, 224)),
transforms.ToTensor(),
transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
])

test_dataset = torchvision.datasets.CIFAR10(root='./data', train=False, download=True, transform=test_transform)
test_loader = torch.utils.data.DataLoader(test_dataset, batch_size=100, shuffle=False)

# Evaluation
model.eval()
correct = 0
total = 0
predictions = []

with torch.no_grad():
for data, target in test_loader:
data, target = data.to(device), target.to(device)
outputs = model(data)
_, predicted = torch.max(outputs.data, 1)
total += target.size(0)
correct += (predicted == target).sum().item()

# Store predictions for submission
predictions.extend(predicted.cpu().numpy())

accuracy = correct / total
print(f'Test Accuracy: {accuracy:.4f} ({100*accuracy:.2f}%)')

# Save predictions to submission.csv
import pandas as pd

# Create probability predictions
model.eval()
all_probs = []

with torch.no_grad():
for data, _ in test_loader:
data = data.to(device)
outputs = model(data)
probs = torch.softmax(outputs, dim=1)
all_probs.extend(probs.cpu().numpy())

# Create submission DataFrame
submission_data = {
'image_id': list(range(len(all_probs)))
}

for i in range(10):
submission_data[f'class_{i}'] = [prob[i] for prob in all_probs]

submission_df = pd.DataFrame(submission_data)
submission_df.to_csv('submission.csv', index=False)
print("Predictions saved to submission.csv")

return accuracy
