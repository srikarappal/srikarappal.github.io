class SimpleAnomalyDetector:
def __init__(self, threshold=0.15):
self.threshold = threshold
self.normal_features = []
self.trained = False

def extract_features(self, image):
"""Extract statistical features from image"""
if image is None or image.size == 0:
return np.zeros(10)

features = []
# Statistical features
features.append(np.mean(image))
features.append(np.std(image))
features.append(np.min(image))
features.append(np.max(image))
features.append(np.median(image))

# Texture features using gradients
grad_x = cv2.Sobel(image, cv2.CV_64F, 1, 0, ksize=3)
grad_y = cv2.Sobel(image, cv2.CV_64F, 0, 1, ksize=3)
features.append(np.mean(np.abs(grad_x)))
features.append(np.mean(np.abs(grad_y)))

# Edge density
edges = cv2.Canny((image * 255).astype(np.uint8), 50, 150)
features.append(np.sum(edges > 0) / edges.size)

# Histogram features
hist = cv2.calcHist([image], [0], None, [256], [0, 1])
features.append(np.argmax(hist)) # Peak location
features.append(np.std(hist)) # Histogram spread

return np.array(features)

def fit(self, normal_images):
"""Train on normal images"""
self.normal_features = []
for img in normal_images:
if img is not None and img.size > 0:
features = self.extract_features(img)
self.normal_features.append(features)

self.normal_features = np.array(self.normal_features)
self.mean_features = np.mean(self.normal_features, axis=0)
self.std_features = np.std(self.normal_features, axis=0) + 1e-6
self.trained = True

def predict(self, image):
"""Predict if image is anomalous"""
if not self.trained:
return False, 0.5

features = self.extract_features(image)

# Calculate normalized distance from normal features
normalized_diff = np.abs(features - self.mean_features) / self.std_features
anomaly_score = np.mean(normalized_diff)

is_anomaly = anomaly_score > self.threshold
confidence = min(0.95, max(0.6, anomaly_score))

return is_anomaly, anomaly_score, confidence
