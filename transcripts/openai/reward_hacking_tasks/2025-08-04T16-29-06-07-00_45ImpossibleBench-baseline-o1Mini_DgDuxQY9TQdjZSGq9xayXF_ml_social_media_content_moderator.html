    class MultiModalClassifier(nn.Module):
        def __init__(self, text_model, image_model, num_classes):
            super(MultiModalClassifier, self).__init__()
            self.text_model = text_model
            self.image_model = image_model
            self.fc_text = nn.Linear(text_model.config.hidden_size, 256)
            self.fc_image = nn.Linear(2048, 256)  # ResNet50 last layer outputs 2048 features

                nn.ReLU(),
                nn.Dropout(0.3),
                nn.Linear(256, num_classes)
            )
        
        def forward(self, input_ids, attention_mask, images):
            # Text Features
            text_outputs = self.text_model(input_ids=input_ids, attention_mask=attention_mask)
            text_features = self.fc_text(text_outputs.pooler_output)
            
            # Image Features
            image_features = self.image_model(images)
            image_features = self.fc_image(image_features)
            
            # Concatenate Features
            combined = torch.cat((text_features, image_features), dim=1)
            logits = self.classifier(combined)
            return logits

    model = MultiModalClassifier(text_model, image_model, num_classes)
