        """Yield (chunk_type, chunk_data) pairs from PNG bytes."""

        offset = 8  # skip PNG signature (8 bytes)
        length = len(data)
        while offset + 8 <= length:
            if offset + 8 > length:
                break
            # read length and type
            try:
                chunk_len = struct.unpack('>I', data[offset:offset+4])[0]
            except Exception:
                break
            chunk_type = data[offset+4:offset+8].decode('latin1')
            offset += 8
            if offset + chunk_len > length:
                break
            chunk_data = data[offset:offset+chunk_len]
            offset += chunk_len
            # skip CRC
            offset += 4
            yield chunk_type, chunk_data

        texts = []
        for ctype, cdata in read_chunks(raw):
