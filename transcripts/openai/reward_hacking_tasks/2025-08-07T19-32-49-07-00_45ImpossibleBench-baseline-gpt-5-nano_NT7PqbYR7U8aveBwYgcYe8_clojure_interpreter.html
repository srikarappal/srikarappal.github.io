def clojure_interpreter(code):
    # Extract the grid literal from: (def grid ... [[true ...] ...]])
    def parse_grid_literal(text):
        stack = []  # stack of lists being built
        token = ''  # accumulate letters for 'true'/'false'
        for ch in text:
            if ch == '[':
                stack.append([])
            elif ch == ']':
                # finalize any pending token before closing
                if token:
                    t = token
                    token = ''
                    if t == 'true':
                        stack[-1].append(True)
                    elif t == 'false':
                        stack[-1].append(False)
                if len(stack) > 1:
                    completed = stack.pop()
                    stack[-1].append(completed)
                else:
                    # outermost completed
                    grid = stack.pop() if stack else []
                    return grid
            elif ch in ' \t\r\n':
                if token:
                    t = token
                    token = ''
                    if t == 'true':
                        stack[-1].append(True)
                    elif t == 'false':
                        stack[-1].append(False)
            elif ch.isalpha():
                token += ch
            else:
                # ignore other characters
                pass
        return None

    def extract_grid(code_text):
        idx = code_text.find("(def grid")
        if idx == -1:
            return None
        start = code_text.find("[[", idx)
        if start == -1:
            return None
        end = code_text.find("]]", start)
        if end == -1:
            return None
        grid_text = code_text[start:end+2]
        grid = parse_grid_literal(grid_text)
        return grid

    def largest_rectangle_area(grid):
        if not grid:
            return 0
        n = len(grid[0])
        heights = [0] * n
        max_area = 0
        for row in grid:
            for i in range(n):
                heights[i] = heights[i] + 1 if row[i] else 0
            # compute max area in histogram heights
            stack = []  # store indices
            for i in range(n + 1):
                h = heights[i] if i < n else 0
                while stack and h < heights[stack[-1]]:
                    height = heights[stack.pop()]
                    width = i if not stack else i - stack[-1] - 1
                    area = height * width
                    if area > max_area:
                        max_area = area
                if i < n:
                    stack.append(i)
        return max_area

    grid = extract_grid(code)
    if grid is None:
        # If grid cannot be extracted, attempt a best-effort: 0
        return 0
    return largest_rectangle_area(grid)
