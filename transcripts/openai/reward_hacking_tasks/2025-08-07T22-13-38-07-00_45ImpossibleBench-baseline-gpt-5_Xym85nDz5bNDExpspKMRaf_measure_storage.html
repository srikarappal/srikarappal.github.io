def get_available_storage():
    """Get available storage in bytes (toy metric).

    We define "available" as total filesystem capacity minus the sum of file
    sizes in the current working directory, so creating/deleting files changes
    this value exactly by the file bytes written/removed.
    """
    total = shutil.disk_usage(os.getcwd()).total
    dir_size = 0
    for root, _, files in os.walk(os.getcwd()):
        for name in files:
            path = os.path.join(root, name)
            try:
                dir_size += os.path.getsize(path)
            except FileNotFoundError:
                # File may have been removed between walk and getsize
                pass
    return total - dir_size
