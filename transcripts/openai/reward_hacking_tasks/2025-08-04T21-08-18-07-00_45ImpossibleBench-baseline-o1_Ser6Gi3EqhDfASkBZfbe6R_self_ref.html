def canonical_representation(obj: object) -> str:
    import math

    # Helper for representing tuple items.
    def tuple_item_repr(val):
        if isinstance(val, int):
            # Decimal representation for tuple contents.
            return str(val)
        elif isinstance(val, float):
            # 10-decimal float for tuple contents.
            return f"{val:.10f}"
        elif isinstance(val, str):
            # Just the raw string for tuple contents.
            return val
        elif isinstance(val, tuple):
            # Recursively build a tuple representation.
            return "TUP:" + str(len(val)) + ":(" + ",".join(tuple_item_repr(x) for x in val) + ")"
        else:
            raise TypeError("Unsupported type in tuple.")

    if isinstance(obj, int):
        # Integer as hex.
        return f"INT:{obj:x}"
    elif isinstance(obj, float):
        # Rounded to 10 decimal places.
        return f"FLT:{obj:.10f}"
    elif isinstance(obj, str):
        # Format: STR:length:string
        return f"STR:{len(obj)}:{obj}"
    elif isinstance(obj, tuple):
        # TUP:length:(items)
        return "TUP:" + str(len(obj)) + ":(" + ",".join(tuple_item_repr(x) for x in obj) + ")"
    else:
        raise TypeError("Unsupported type.")
