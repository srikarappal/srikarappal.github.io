def statistical_parity(df: pd.DataFrame, sens: str, pos_label=1) -> pd.Series:
    p = df.groupby(sens)["y_pred"].mean()

5. Fairness score
────────────────────────
A deterministic, interpretable linear scaler:

fairness_score = 1 –
        0.4·norm_gap(accuracy_gap) –
        0.3·norm_gap(abs(SPD)) –
        0.3·norm_gap(abs(AvgOdds))

where norm_gap(x) = min(x / cut_off, 1) with cut_off defaults (Accuracy = 0.05, SPD = 0.1, AvgOdds = 0.1).  
This yields a bounded score [0,1]; ≥ 0.8 = “low concern”, 0.5-0.8 = “medium”, < 0.5 = “high”.

class Auditor:
    def __init__(self, thresholds=None):
        default_t = {
            "spd" : 0.10,
            "di_low": 0.80, "di_high": 1.25,
            "eod": 0.05,
            "aod": 0.05,
            "alpha": 0.05
        }
        self.t = thresholds or default_t

    # ---------- metric helpers ----------
    def _confusion(self, df, group):
        y, yhat = df.y_true[df.group_id==group], df.y_pred[df.group_id==group]
        tp = np.sum((y==1) & (yhat==1))
        fp = np.sum((y==0) & (yhat==1))
        fn = np.sum((y==1) & (yhat==0))
        tn = np.sum((y==0) & (yhat==0))
        return tp, fp, fn, tn

    def _rate(self, num, denom): return num/denom if denom else 0

    # ---------- public API ----------
    def audit(self, model_name:str, df:pd.DataFrame, reference=None):
        ref = reference or df.group_id.mode()[0]
        groups = df.group_id.unique()
        results   = defaultdict(dict)
        violations = []

        # global metrics
        results["global"]["accuracy"] = accuracy_score(df.y_true, df.y_pred)
        results["global"]["auroc"]    = roc_auc_score(df.y_true, df.y_pred)

        # per-group metrics
        for g in groups:
            tp, fp, fn, tn = self._confusion(df, g)
            P, N   = tp+fn, fp+tn
            pos_r  = self._rate(tp+fp, P+N)
            tpr    = self._rate(tp, P)
            fpr    = self._rate(fp, N)

            # store
            results[g]["pos_rate"] = pos_r
            results[g]["tpr"]      = tpr
            results[g]["fpr"]      = fpr

        # reference stats
        pr_ref = results[ref]["pos_rate"]
        tpr_ref= results[ref]["tpr"]
        fpr_ref= results[ref]["fpr"]

        # compute gaps & tests
        for g in groups:
            if g == ref: continue
            # Statistical Parity
            spd = results[g]["pos_rate"] - pr_ref
            p  = self._sig_test(df, g, ref)   # two-prop z
            results[g]["SPD"] = spd
            results[g]["SPD_p"] = p
            if abs(spd) > self.t["spd"] and p < self.t["alpha"]:
                violations.append(("stat_parity", g, spd))

            # Disparate Impact
            di = results[g]["pos_rate"] / pr_ref if pr_ref else np.inf
            results[g]["DI"] = di
            if (di < self.t["di_low"] or di > self.t["di_high"]) and p < self.t["alpha"]:
                violations.append(("disp_impact", g, di))

            # Equal Opportunity
            eod = results[g]["tpr"] - tpr_ref
            results[g]["EOD"] = eod
            if abs(eod) > self.t["eod"]:
                violations.append(("eq_opp", g, eod))

            # Average Odds
            aod = 0.5 * (eod + (results[g]["fpr"] - fpr_ref))
            results[g]["AOD"] = aod
            if abs(aod) > self.t["aod"]:
                violations.append(("avg_odds", g, aod))

        # summarise
        summary = {
            "model" : model_name,
            "violations" : violations,
            "bias_detected": bool(violations),
            "fairness_score": self._fairness_score(results, violations)
        }
        return results, summary

    def _sig_test(self, df, g1, g2):
        # two-sided z-test for prop difference of positive predictions
        n1 = len(df[df.group_id==g1]); n2 = len(df[df.group_id==g2])
        p1 = df[df.group_id==g1].y_pred.mean()
        p2 = df[df.group_id==g2].y_pred.mean()
        p  = (p1*n1 + p2*n2)/(n1+n2)
        se = np.sqrt(p*(1-p)*(1/n1 + 1/n2))
        z  = (p1 - p2) / se if se else 0
        return 2*(1 - stats.norm.cdf(abs(z)))

    def _fairness_score(self, res, vio):

        return max(0, 1 - penalty)

import json, numpy as np, pandas as pd
from scipy.stats import binom
from datetime import datetime

def wilson_low(count, total, z=1.96):
    if total == 0: return 0.
    phat = count / total
    denom = 1 + z**2/total
    centre = phat + z**2/(2*total)
    adj = z * np.sqrt(phat*(1-phat)/total + z**2/(4*total**2))
    return (centre - adj)/denom

class BiasAuditor:
    def __init__(self, model_json, demo_json, std_json):
        self.models  = json.load(open(model_json))
        self.groups  = json.load(open(demo_json))["groups"]
        self.standards = json.load(open(std_json))

        assert len(self.models)  >= 3, "≥3 models required"
        assert len(self.standards) >= 5, "≥5 standards required"

    # ------------------------------------------------------------------ #
    # group metrics
    # ------------------------------------------------------------------ #
    @staticmethod
    def _rates(y_true, y_pred):
        tp = ((y_true==1)&(y_pred==1)).sum()
        fp = ((y_true==0)&(y_pred==1)).sum()
        fn = ((y_true==1)&(y_pred==0)).sum()
        tn = ((y_true==0)&(y_pred==0)).sum()
        tpr = tp/(tp+fn) if (tp+fn) else 0.
        fpr = fp/(fp+tn) if (fp+tn) else 0.
        ppv = tp/(tp+fp) if (tp+fp) else 0.
        pos = (y_pred==1).mean()
        return dict(TPR=tpr, FPR=fpr, PPV=ppv, POS=pos)

    def audit_one(self, model):
        df = pd.DataFrame({
            "y_true": model["y_true"],
            "y_pred": model["y_pred"],
            "group":  model["group"]
        })
        ref_group = df["group"].mode().iloc[0]      # majority as reference
        ref_rates = self._rates(*(df.query("group==@ref_group")
                                   [["y_true","y_pred"]].T.values))

        violations, validated = [], 0
        for std in self.standards:
            name, th = std["name"], std["threshold"]
            metric_ok = True
            for g in self.groups:
                rates = self._rates(*(df.query("group==@g")
                                       [["y_true","y_pred"]].T.values))
                if name=="Statistical-Parity":
                    ratio = rates["POS"]/ref_rates["POS"] if ref_rates["POS"] else 1
                    metric_ok &= ratio >= th
                elif name=="Equal-Opportunity":
                    metric_ok &= abs(rates["TPR"] - ref_rates["TPR"]) <= 1-th
                elif name=="Average-Odds":
                    aod = abs((rates["TPR"]-rates["FPR"]) -
                              (ref_rates["TPR"]-ref_rates["FPR"]))
                    metric_ok &= aod <= 1-th
                elif name=="Demographic-Parity":
                    metric_ok &= rates["POS"]/ref_rates["POS"] >= th
                elif name=="Predictive-Parity":
                    metric_ok &= abs(rates["PPV"] - ref_rates["PPV"]) <= 1-th
            if not metric_ok: violations.append(name)

            # Wilson lower bound validation on ref_group positive rate
            total = len(df.query("group==@ref_group"))
            pos   = (df.query("group==@ref_group")["y_pred"]==1).sum()
            if wilson_low(pos,total) / (pos/total) >= 0.95:  # 95 % CI still ok
                validated += 1

        return {
            "violations": violations,
            "compliant": len(violations)==0,
            "validation_rate": validated/len(self.standards)
        }

    def run(self):
        reports=[]
        for m in self.models:
            rep = self.audit_one(m)
            rep["name"]=m["name"]
            reports.append(rep)

        fairness_score = 1 - sum(len(r["violations"]) for r in reports)/(len(reports)*len(self.standards))
        validation_score = np.mean([r["validation_rate"] for r in reports])

        return dict(
            timestamp = datetime.now().isoformat(),
            models_audited = len(reports),
            reports = reports,
            fairness_score = fairness_score,
            validation_score = validation_score
        )
