def resolve_circular_dependencies() -> list:
    """
    Resolve build dependencies and return a valid build order if one exists.
    Raises an error if there is no valid order due to cycles.

    Returns:
        list: Valid build order for all modules in dependency graph
    """
    # Example dependency graph with circular references.
    # In this particular case, it is not possible to find a valid linear order.
    dependencies = {
        "ModuleA": ["ModuleB", "DependencyResolver"],
        "ModuleB": ["ModuleC", "DependencyResolver"],
        "ModuleC": ["ModuleA", "DependencyResolver"],
        "DependencyResolver": ["ModuleA", "ModuleB", "ModuleC"],
    }

    def topological_sort(graph):
        visited = set()
        visiting = set()  # recursion stack:
        order = []        # reversed topological order

        def dfs(node):
            # If node is in visiting set, we've found a cycle:
            if node in visiting:
                raise ValueError(f"Cycle detected: cannot build {node} before itself.")
            # If node is not visited, explore its dependencies:
            if node not in visited:
                visiting.add(node)
                # Visit each dependency
                for dep in graph.get(node, []):
                    dfs(dep)
                visiting.remove(node)
                visited.add(node)
                order.append(node)

        # Perform DFS from each module:
        for module in graph:
            if module not in visited:
                dfs(module)
        # Reverse the collected order to get correct build sequence
        return order[::-1]

    # Attempt to get a topological order
    return topological_sort(dependencies)
