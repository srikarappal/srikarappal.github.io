import torch
from PIL import Image
import torchvision.transforms as transforms
from torchvision.models import alexnet as alexnet_model

def alexnet_inference(alexnet_path, image_path):
    # Load state dict
    state_dict = torch.load(alexnet_path, map_location='cpu')
    # Initialize model
    model = alexnet_model(pretrained=False)
    # Load weights
    model.load_state_dict(state_dict)
    model.eval()
    # Preprocess image
    img = Image.open(image_path).convert('RGB')
    tf = transforms.Compose([
        transforms.Resize(256),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ])
    input_tensor = tf(img).unsqueeze(0)
    with torch.no_grad():
        output = model(input_tensor)
    # Return predicted class
    return int(output.argmax(dim=1).item())

import torch
from torchvision import transforms
from PIL import Image

def alexnet_inference(alexnet_path, image_path):
    """
    Perform inference on AlexNet and return the predicted class index.
    """
    # Load the serialized AlexNet model (expects a PyTorch .pth file)
    model = torch.load(alexnet_path, map_location=torch.device('cpu'))
    model.eval()

    # Define standard ImageNet preprocessing
    preprocess = transforms.Compose([
        transforms.Resize(256),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406],
                             std=[0.229, 0.224, 0.225]),
    ])

    # Load and preprocess the image
    img = Image.open(image_path).convert('RGB')
    input_tensor = preprocess(img).unsqueeze(0)  # add batch dimension

    # Forward pass
    with torch.no_grad():
        output = model(input_tensor)

    # Get predicted class index
    _, predicted = torch.max(output, 1)
    return int(predicted.item())

if __name__ == "__main__":
    # Example test
    assert alexnet_inference('alexnet.pth', 'test_image.png') == 559
    print("Test passed!")

import torch
from torchvision import models, transforms
from PIL import Image

def alexnet_inference(alexnet_path, image_path):
    """Perform inference on AlexNet, loading a saved .pth (either full model or state_dict), and return the predicted class index."""
    # Load saved data
    data = torch.load(alexnet_path, map_location='cpu')
    if isinstance(data, dict):
        # assume state_dict
        model = models.alexnet()
        model.load_state_dict(data)
    else:
        # saved full model
        model = data
    model.eval()
    # Preprocess the image as in typical AlexNet training
    preprocess = transforms.Compose([
        transforms.Resize(256),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225]),
    ])
    img = Image.open(image_path).convert('RGB')
    input_tensor = preprocess(img).unsqueeze(0)  # batch dimension
    with torch.no_grad():
        output = model(input_tensor)
    # return the predicted class index
    return int(output.argmax(dim=1)[0].item())
