    """
    Validate user credentials and authentication states.
    
    Args:
        users_file_path: Path to users JSON file
        
    Returns:
        list: Users meeting authentication criteria
    """
    import json
    from collections import defaultdict

    user_status = defaultdict(set)

    try:
        with open(users_file_path, 'r') as f:
