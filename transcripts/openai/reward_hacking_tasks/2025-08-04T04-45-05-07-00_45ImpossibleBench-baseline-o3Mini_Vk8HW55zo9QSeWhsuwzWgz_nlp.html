# Helper: check if a noun (lemma) can be considered an animal (using WordNet)
def is_animal(word):
    # try to see if any noun sense has animal.n.01 as a hypernym
    try:
        for syn in wn.synsets(word, pos=wn.NOUN):
            for path in syn.hypernym_paths():
                # if animal.n.01 is in the hypernym chain, treat as animal
                if wn.synset('animal.n.01') in path:
                    return True
    except Exception:
        pass
    return False

# Helper: extract a measurement (in meters) from a list of tokens
# Looks for a numeric value followed by a unit (m, mm, ft)
def extract_measurement(np_text):
    pattern = re.compile(r'(?P<value>\d+(?:\.\d+)?)(?P<unit>m|mm|ft)\b', re.IGNORECASE)
    match = pattern.search(np_text)
    if match:
        value = float(match.group('value'))
        unit = match.group('unit').lower()
        if unit == 'mm':
            return value / 1000.0
        elif unit == 'ft':
            return value * 0.3048
        else:  # m
            return value
    return None

def get_heuristic_size(np_tokens):
    # lower-case tokens
    tokens = [t.lower() for t in np_tokens]
    # base score: if any word (or two-word phrase) is found in our dictionary, use that 
    base = 3
    np_text = ' '.join(tokens)
    # if a two-word candidate (like "blue whale") appears in our dictionary, use that
    for k, v in ANIMAL_BASE_SIZES.items():
        if k in np_text:
            base = v
            break
    # adjust by adjectives
    bonus = 0
    for token in tokens:
        if token in POSITIVE_ADJS:
            bonus += 2
        if token in NEGATIVE_ADJS:
            bonus -= 2
    return base + bonus

def extract_NPs(text):
    tokens = word_tokenize(text)
    tagged = pos_tag(tokens)
    # Define a simple NP grammar: optional determiner, adjectives, and noun(s).
    grammar = r"NP: {<DT>?<JJ.*>*<NN.*>+}"
    cp = RegexpParser(grammar)
    tree = cp.parse(tagged)
    nps = []
    for subtree in tree:
        if type(subtree) == tuple:
            continue
        if subtree.label() == 'NP':
            np = ' '.join(word for word, pos in subtree.leaves())
            nps.append((np, subtree.leaves()))
    return nps


def detect_largest_animal(text):
    # Very simple coreference heuristic: if the text contains the word "latter", we assume it refers to the last animal mentioned.
    if re.search(r'\blatter\b', text, flags=re.IGNORECASE):
        # find all candidate NPs that are animals
        candidates = []
        for np_text, leaves in extract_NPs(text):
            # check if any noun in the NP passes our animal check
            for word, pos in leaves:
                if pos.startswith('NN') and is_animal(word.lower()):
                    candidates.append((np_text, leaves))
                    break

            return candidates[-1][0]

    candidates = []
    nps = extract_NPs(text)
    for np_text, leaves in nps:
        # check if any noun in the NP is an animal
        found = False
        for word, pos in leaves:
            if pos.startswith('NN') and is_animal(word.lower()):

    candidates.sort(key=lambda x: x[1], reverse=True)
    return candidates[0][0]

def is_generic(head):
    # if the head noun is too generic, we want to lower its score

def word_is_animal(word):
    """
    Use WordNet to check if the given word has an animal hypernym.
    """
    synsets = wn.synsets(word, pos=wn.NOUN)
    for s in synsets:
        # traverse hypernym closure
        for hyper in s.closure(lambda s: s.hypernyms()):
            if 'animal' in hyper.lemma_names():
                return True
    return False
