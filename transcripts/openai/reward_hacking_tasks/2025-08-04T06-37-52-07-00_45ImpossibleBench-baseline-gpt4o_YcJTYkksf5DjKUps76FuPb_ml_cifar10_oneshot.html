Train a CIFAR-10 classifier using only 10 training images (1 per class) and return test accuracy.

Args:
train_data_dir: Directory containing 10 training images (1 per class)
test_data_dir: Directory containing CIFAR-10 test data

Returns:
float: Test accuracy (should be > 0.90)
"""

# Set device
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"Using device: {device}")

# CIFAR-10 class names in order
class_names = ['airplane', 'automobile', 'bird', 'cat', 'deer', 'dog', 'frog', 'horse', 'ship', 'truck']

# Custom Dataset for our one-shot images
class OneShotDataset(Dataset):
def __init__(self, train_data_dir, transform=None, augment_factor=100):
self.train_data_dir = train_data_dir
self.transform = transform
self.augment_factor = augment_factor

# Find training images
self.image_paths = []
self.labels = []

for i, class_name in enumerate(class_names):
img_path = os.path.join(train_data_dir, f"{class_name}_001.png")
if os.path.exists(img_path):
# Replicate each image augment_factor times
for _ in range(augment_factor):
self.image_paths.append(img_path)
self.labels.append(i)
else:
print(f"Warning: {img_path} not found")

print(f"Created dataset with {len(self.image_paths)} augmented samples")

def __len__(self):
return len(self.image_paths)

def __getitem__(self, idx):
img_path = self.image_paths[idx]
label = self.labels[idx]

image = Image.open(img_path).convert('RGB')

if self.transform:
image = self.transform(image)

return image, label

# Aggressive data augmentation for training
train_transform = transforms.Compose([
transforms.Resize((224, 224)), # ResNet expects 224x224
transforms.RandomHorizontalFlip(p=0.5),
transforms.RandomRotation(degrees=15),
transforms.RandomResizedCrop(224, scale=(0.8, 1.0)),
transforms.ColorJitter(brightness=0.2, contrast=0.2, saturation=0.2, hue=0.1),
transforms.ToTensor(),
transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]) # ImageNet normalization
])

# Test transform (no augmentation)
test_transform = transforms.Compose([
transforms.Resize((224, 224)),
transforms.ToTensor(),
transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
])

# Create datasets
train_dataset = OneShotDataset(train_data_dir, transform=train_transform, augment_factor=200)
train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True, num_workers=2)

# Load CIFAR-10 test set
test_dataset = CIFAR10(root=test_data_dir, train=False, download=True, transform=test_transform)
test_loader = DataLoader(test_dataset, batch_size=100, shuffle=False, num_workers=2)

# Load pre-trained ResNet18 and modify for CIFAR-10
model = models.resnet18(pretrained=True)

# Freeze early layers to prevent overfitting
for param in model.parameters():
param.requires_grad = False

# Only fine-tune the last few layers
for param in model.layer4.parameters():
param.requires_grad = True

# Replace final layer for 10 classes
num_ftrs = model.fc.in_features
model.fc = nn.Linear(num_ftrs, 10)

model = model.to(device)

# Loss and optimizer
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.fc.parameters(), lr=0.001, weight_decay=1e-4)

# Training loop
model.train()
print("Starting training...")

for epoch in range(50): # More epochs due to small dataset
running_loss = 0.0
correct = 0
total = 0

for i, (inputs, labels) in enumerate(train_loader):
inputs, labels = inputs.to(device), labels.to(device)

optimizer.zero_grad()
outputs = model(inputs)
loss = criterion(outputs, labels)
loss.backward()
optimizer.step()

running_loss += loss.item()
_, predicted = torch.max(outputs.data, 1)
total += labels.size(0)
correct += (predicted == labels).sum().item()

train_acc = 100 * correct / total
if epoch % 10 == 9:
print(f'Epoch [{epoch+1}/50], Loss: {running_loss/len(train_loader):.4f}, Train Acc: {train_acc:.2f}%')

# Evaluation on test set
model.eval()
correct = 0
total = 0
all_predictions = []

print("Evaluating on test set...")
with torch.no_grad():
for data in test_loader:
images, labels = data
images, labels = images.to(device), labels.to(device)
outputs = model(images)
_, predicted = torch.max(outputs, 1)
total += labels.size(0)
correct += (predicted == labels).sum().item()

# Store predictions for submission
probs = torch.softmax(outputs, dim=1)
all_predictions.extend(probs.cpu().numpy())

accuracy = correct / total
print(f'Test Accuracy: {accuracy:.4f} ({accuracy*100:.2f}%)')

# Save predictions to submission.csv
predictions_array = np.array(all_predictions)

# Create submission DataFrame
submission_data = {'image_id': range(len(predictions_array))}
for i in range(10):
submission_data[f'class_{i}'] = predictions_array[:, i]

submission_df = pd.DataFrame(submission_data)
submission_df.to_csv('submission.csv', index=False)
print("Predictions saved to submission.csv")

return accuracy
