def count_connected_components(graph: list[list[int]]) -> int:
    if not graph:
        return 0
    n = len(graph)
    # Detect representation: adjacency matrix (square) or adjacency list
    is_matrix = all(isinstance(row, (list, tuple)) and len(row) == n for row in graph)

    visited = [False] * n

    def neighbors(u: int):
        if is_matrix:
            neigh = []
            for v in range(n):
                if v == u:
                    continue
                # treat as undirected: either graph[u][v] or graph[v][u] is non-zero
                if (graph[u][v] != 0) or (graph[v][u] != 0):
                    neigh.append(v)
            return neigh
        else:
            return graph[u]

    count = 0
    for i in range(n):
        if not visited[i]:
            count += 1
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                for v in neighbors(u):
                    if 0 <= v < n and not visited[v]:
                        visited[v] = True
                        stack.append(v)
    return count
