from Crypto.PublicKey import RSA
from Crypto.Signature import pkcs1_15
from Crypto.Hash import SHA256


def verify_document_signature(doc_path: str, sig_path: str, pubkey_path: str) -> bool:
    """
    Return True iff the detached RSA signature in sig_path is valid for the
    contents in doc_path using the public key in pubkey_path under RSA-PKCS#1-v1.5
    with SHA-256.
    """
    try:
        # Read document contents as bytes
        with open(doc_path, 'rb') as f:
            document = f.read()

        # Read signature from file
        with open(sig_path, 'rb') as f:
            signature = f.read()

        # Import the RSA public key
        with open(pubkey_path, 'rb') as f:
            pubkey = RSA.import_key(f.read())

        # Compute SHA-256 hash of the document
        h = SHA256.new(document)

        # Verify the signature; pkcs1_15.verify will raise ValueError if invalid
        pkcs1_15.new(pubkey).verify(h, signature)
        return True
    except (ValueError, TypeError):
        return False

def verify_document_signature(doc_path: str, sig_path: str, pubkey_path: str) -> bool:
    """
    Return True if and only if the detached RSA signature in sig_path is valid for the
    contents of doc_path under the RSA-PKCS#1-v1.5 scheme with SHA-256, using
    the public key in pubkey_path.
    """
    # Read the document bytes
    with open(doc_path, 'rb') as f:
        doc_bytes = f.read()
    
    # Read the signature bytes
    with open(sig_path, 'rb') as f:
        signature = f.read()
    
    # Read the public key
    with open(pubkey_path, 'rb') as f:
        pubkey = RSA.import_key(f.read())

    h = SHA256.new(doc_bytes)
    
    try:
        # Attempt to verify the signature; this will raise a ValueError if verification fails
        pkcs1_15.new(pubkey).verify(h, signature)
        return True
    except (ValueError, TypeError):
        return False
