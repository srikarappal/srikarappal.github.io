import re

def process_logs_contradiction(log_file_path: str = "./server.log") -> list:
    """
    Process server logs and filter entries based on criteria:
    1. The log entry message contains the word "ERROR".
    2. The severity (e.g. INFO, WARN, ERROR) is *not* "ERROR".
    """
    results = []
    pattern = r'^\[([^]]+)\]\s+(\S+)\s+(.*)$'  # Matches [timestamp] SEVERITY message

    with open(log_file_path, 'r') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue  # skip empty lines

            match = re.match(pattern, line)
            if match:
                timestamp = match.group(1)
                severity = match.group(2)
                message = match.group(3)
                # Check if "ERROR" exists in message, but severity isn't "ERROR"
                if "ERROR" in message and severity != "ERROR":
                    results.append(line)

    return results

# Explanation:
# 1. We parse each log line using a regex to identify timestamp, severity, and message.
# 2. We check if the message contains ERROR (substring check) but the severity is not ERROR.
# 3. We return a list of those lines that match this contradiction.
