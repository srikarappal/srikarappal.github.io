    Validate user credentials and authentication states by finding users

    import json

    def normalize_username(value: Any) -> str:
        if value is None:
            return ""
        s = str(value).strip()
        # If email, take the part before '@'
        if "@" in s:
            s = s.split("@", 1)[0]
        return s.lower()

    def extract_username(rec: Dict[str, Any]) -> str:
        # Prefer common username fields
        for key in ("username", "user"):
            if key in rec and rec[key]:
                return normalize_username(rec[key])
        # Fallback to email prefix as username
        if "email" in rec and rec["email"]:
            return normalize_username(rec["email"])
        return ""

    def is_auth_key(key: str) -> bool:
        k = key.lower()
        # Only treat clearly auth-related keys as signals
        return (
            "auth" in k or
            "login" in k or
            "logged" in k
        )

    TRUTHY_STR = {"true", "yes", "y", "1", "authenticated", "logged_in", "loggedin", "active", "online"}
    FALSY_STR = {"false", "no", "n", "0", "unauthenticated", "logged_out", "loggedout", "inactive", "offline", "signed_out", "signout", "signedout"}

    def str_to_bool(s: str):
        ls = s.strip().lower()
        if ls in TRUTHY_STR:
            return True
        if ls in FALSY_STR:
            return False
        return None

    def collect_states_from_obj(obj: Any) -> Iterable[bool]:
        """Yield authentication states found within obj (dict/list/primitive)."""
        if isinstance(obj, dict):
            for k, v in obj.items():

                if k.lower() in {"sessions", "session"}:
                    yield from collect_states_from_obj(v)

                        if isinstance(v, bool):
                            yield v
                        elif isinstance(v, str):
                            bv = str_to_bool(v)
                            if bv is not None:
                                yield bv
                        # If nested structure under an auth-related key, recurse

            yield obj

            bv = str_to_bool(obj)
            if bv is not None:
                yield bv
