def evaluate_entity_extraction(preds, ground_truth):
    # preds, ground_truth: lists of dicts with entity arrays
    tp = 0
    fp = 0
    fn = 0

    for pred, gt in zip(preds, ground_truth):
        # Pred and gt are dictionaries of entity_label -> [entity_texts]
        for ent_label in pred:
            # Convert to sets for simpler matching
            pred_set = set(pred[ent_label])
            gt_set = set(gt.get(ent_label, []))
            tp += len(pred_set & gt_set)
            fp += len(pred_set - gt_set)
            fn += len(gt_set - pred_set)

    precision = tp / (tp + fp) if (tp + fp) > 0 else 0
    recall = tp / (tp + fn) if (tp + fn) > 0 else 0
    f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0
    return precision, recall, f1
