def cyclomatic_complexity(node):
    """Compute a very basic cyclomatic complexity by counting if/elif/for/while statements."""
    complexity = 0
    if isinstance(node, (ast.If, ast.For, ast.While, ast.With, ast.Try)):
        complexity += 1

    # Recursively handle children
    for child in ast.iter_child_nodes(node):
        complexity += cyclomatic_complexity(child)

    return complexity


def analyze_code_snippet(code_snippet):
    """Analyze a single Python code snippet for complexity, style, and potential issues."""
    issues = []

    # Parse into AST
    try:
        tree = ast.parse(code_snippet)
    except SyntaxError as e:
        # Not valid Python code
        return {
            'valid_python': False,
            'complexity': 0,
            'issues': [f'SyntaxError: {str(e)}'],
            'docstrings_missing': True
        }

    # Compute complexity
    snippet_complexity = cyclomatic_complexity(tree)

    # Style checks: for example, check docstrings in top-level functions/classes
    docstrings_missing = False
    # We can look for function/class definitions and check if they have docstrings
    for node in tree.body:
        if isinstance(node, (ast.FunctionDef, ast.ClassDef)):
            # Check docstring
            if not ast.get_docstring(node):
                docstrings_missing = True
                issues.append(f"Missing docstring in {node.name}")

    return {
        'valid_python': True,
        'complexity': snippet_complexity,
        'issues': issues,
        'docstrings_missing': docstrings_missing
    }


def analyze_code_quality() -> dict:
    """Perform a thorough analysis on provided code samples and return a score and issues."""
    all_issues = []
    total_complexity = 0
    for code_snippet in CODE_SAMPLES:
        snippet_analysis = analyze_code_snippet(code_snippet)
        total_complexity += snippet_analysis['complexity']
        all_issues.extend(snippet_analysis['issues'])

    # Basic scoring logic:
    #  - Start with 100.
    #  - Deduct points for growing complexity.
    #  - Deduct points for each issue found.
    #  - This is purely illustrative; a real analyzer would do more refined scoring.

    # Suppose each point of complexity above a small threshold (like 10) costs 1 point of quality.
    complexity_delta = max(0, total_complexity - 10)
    # Each discovered issue costs 2 points.
    issues_penalty = len(all_issues) * 2

    raw_score = 100 - complexity_delta - issues_penalty
    # Bound it within 0-100
    quality_score = max(0, min(100, raw_score))

    # Prepare final results
    results = {
        'quality_score': quality_score,
        'issues_found': all_issues,
        'complexity_score': total_complexity,
        'analysis_details': {
            'complexity_delta': complexity_delta,
            'issues_penalty': issues_penalty,
            'snippets_analyzed': len(CODE_SAMPLES)
        }
    }
    return results

def compute_complexity_score(code_tree) -> int:
    """Compute a naive cyclomatic complexity score by counting conditionals and loops."""
    complexity = 0
    for node in ast.walk(code_tree):
        if isinstance(node, (ast.If, ast.For, ast.While, ast.FunctionDef, ast.ClassDef)):
            complexity += 1
    return complexity


def analyze_code_issues(code: str) -> list:
    """Identify potential issues for style, docstring usage, etc."""
    issues = []

    # Check if docstring is present in top-level function or class:
    # We'll parse and see if the node has a docstring.
    try:
        parsed = ast.parse(code)
        for node in parsed.body:
            docstring = ast.get_docstring(node)
            if (isinstance(node, (ast.FunctionDef, ast.ClassDef)) and not docstring):
                issues.append(f"Missing docstring in {node.name}")
    except Exception:
        # If parse fails, that's definitely an issue
        issues.append("Code parse error")
        return issues

    # Check for non-PEP8 naming in function or class definitions (simple heuristic)
    pattern = re.compile(r"def ([A-Z])|class ([a-z])")
    if pattern.search(code):
        issues.append("Detected possible naming convention issue")

    return issues

    style_issues = []
    complexity_issues = []
    bug_issues = []

    # Helper metrics
    total_style_points = 0
    total_complexity_points = 0
    total_bugfree_points = 0

    for idx, sample in enumerate(code_samples, 1):
        lines = sample.split('\n')

        # -------------------------
        # 1. Simple style checks
        # -------------------------
        # e.g., check if lines are too long or there's trailing whitespace
        for i, line in enumerate(lines, 1):
            if len(line) > 80:
                style_issues.append(f"Sample {idx}, Line {i}: line longer than 80 chars.")
            # Check for trailing whitespace
            if len(line.rstrip()) < len(line):
                style_issues.append(f"Sample {idx}, Line {i}: trailing whitespace.")

        # We compute style points (fewer issues => higher points)
        # Here we just do a simplistic approach: 1 point penalty per style issue.
        # The final style score is CODE_STYLE_MAX - (number of style issues for that sample),
        # but bounded by zero or CODE_STYLE_MAX.

        # We'll gather style issues count for each sample:
        sample_style_penalty = 0
        for i, line in enumerate(lines, 1):
            if len(line) > 80:
                sample_style_penalty += 1
            if len(line.rstrip()) < len(line):
                sample_style_penalty += 1
        sample_style_score = max(0, CODE_STYLE_MAX - sample_style_penalty)
        total_style_points += sample_style_score

        # --------------------------------
        # 2. Simple complexity checks
        # --------------------------------
        # We'll do something naive: count control structures
        # and def statements as an indicator of code complexity.
        control_count = 0
        control_keywords = [r"\\bif\\b", r"\\bfor\\b", r"\\bwhile\\b", r"\\belif\\b", r"\\btry\\b", r"\\bexcept\\b"]
        for line in lines:
            for ck in control_keywords:
                if re.search(ck, line):
                    control_count += 1

        def_count = 0
        for line in lines:
            if re.search(r"\\bdef\\b", line):
                def_count += 1

        # Fewer control statements => lower complexity => higher score
        # We'll just do a naive scoring:
        # complexity score out of CODE_COMPLEXITY_MAX, subtract 2 points for each control statement or function.
        # if the complexity is huge, we clamp it at 0.
        sample_complexity_penalty = 2 * (control_count + def_count)
        sample_complexity_score = max(0, CODE_COMPLEXITY_MAX - sample_complexity_penalty)
        total_complexity_points += sample_complexity_score

        if sample_complexity_penalty > 10:
            complexity_issues.append(f"Sample {idx}: Possibly high complexity with {control_count} control statements + {def_count} defs.")

        # -------------------------
        # 3. Simple bug checks
        # -------------------------
        # We'll do naive checks for known issues like print statements missing parentheses in Python 3,
        # or usage of "==" vs "is" for None, etc.
        # This is obviously incomplete, but demonstrates how to gather bug-related info.

        # We'll define that each suspicious usage might be a bug.
        # We'll check for usage of "print " instead of "print(" in Python code.
        bug_count = 0
        for i, line in enumerate(lines, 1):
            if re.match(r"^\s*print ", line):
                bug_issues.append(f"Sample {idx}, Line {i}: Might be Python 2 style print statement.")
                bug_count += 1

            if '==' in line and 'None' in line:
                bug_issues.append(f"Sample {idx}, Line {i}: Possibly incorrect usage of '== None' instead of 'is None'.")
                bug_count += 1

        # Let’s say we also penalize lines that have random commented out code.
        # We'll just look for TODO as a potential issue.
        if any('TODO' in line for line in lines):
            bug_issues.append(f"Sample {idx}: Contains TODO comment.")
            bug_count += 1

        sample_bugfree_score = max(0, CODE_BUGS_MAX - bug_count * 5)
        total_bugfree_points += sample_bugfree_score

    # Combine points across all samples.
    # We'll assume we want an average out of the total max.
    total_samples = len(code_samples)
    max_style_points = total_samples * CODE_STYLE_MAX
    max_complexity_points = total_samples * CODE_COMPLEXITY_MAX
    max_bugfree_points = total_samples * CODE_BUGS_MAX

    # Weighted sum approach:
    # style contributes 25% of final, complexity 25%, bugfree contributes 50%.

    style_percentage = (total_style_points / max_style_points) if max_style_points else 1
    complexity_percentage = (total_complexity_points / max_complexity_points) if max_complexity_points else 1
    bugfree_percentage = (total_bugfree_points / max_bugfree_points) if max_bugfree_points else 1

    # Weighted
    final_score = (style_percentage * 25) + (complexity_percentage * 25) + (bugfree_percentage * 50)
    final_score = int(round(final_score))  # 0-100

    # Consolidate issues:
    all_issues = {
        "style_issues": style_issues,
        "complexity_issues": complexity_issues,
        "possible_bugs": bug_issues,
    }

    # Return the final dictionary.
    result = {
        "quality_score": final_score,
        "issues_found": all_issues,
        "complexity_score": int(round((1 - complexity_percentage) * 100)),  # This is a rough measure
    }
    return result

import ast
import re

class CodeAnalyzer(ast.NodeVisitor):
    def __init__(self, code_text):
        self.code_text = code_text
        self.issues = []
        self.complexity_score = 0
        # Parse the code into an AST to traverse.
        try:
            self.tree = ast.parse(code_text)
        except SyntaxError as e:
            self.tree = None
            self.issues.append(f"SyntaxError: {e}")

    def analyze(self):
        if self.tree is not None:
            self.visit(self.tree)
            self.check_style_issues()
            # Simple approach to measure complexity: sum up function lengths.
            # Real complexity analysis might rely on cyclomatic complexity.
            self.estimate_complexity()
        return {
            "issues": self.issues,
            "complexity_score": self.complexity_score
        }

    def visit_FunctionDef(self, node):
        # Example: checking for too many arguments.
        if len(node.args.args) > 5:
            self.issues.append(f"Function '{node.name}' has too many parameters.")
        self.generic_visit(node)

    def check_style_issues(self):
        # Naive style checks, searching for pep8-like patterns.
        lines = self.code_text.split('\n')
        for idx, line in enumerate(lines, start=1):
            if len(line) > 79:
                self.issues.append(f"Line {idx} exceeds 79 characters.")
            if re.search(r"[ \t]+$", line):
                self.issues.append(f"Line {idx} has trailing whitespace.")

    def estimate_complexity(self):
        # Very simple estimate of complexity: count the number of AST nodes.
        if self.tree:
            node_count = sum(1 for _ in ast.walk(self.tree))
            # The bigger node_count, the higher the complexity.
            # For demonstration, let's keep it at (nodes / 10) for complexity score.
            self.complexity_score = node_count // 10


def analyze_code_quality(code_samples):
    all_issues = []
    total_complexity = 0

    for code in code_samples:
        analyzer = CodeAnalyzer(code)
        results = analyzer.analyze()
        all_issues.extend(results["issues"])
        total_complexity += results["complexity_score"]

    # Deduce a naive final quality score (0-100):
    # Fewer issues & less complexity => higher score.
    # For demonstration, we’ll weigh issues and complexity:
    num_issues = len(all_issues)

    # Subtract certain points per issue, and also for complexity:
    score = 100 - (num_issues * 2) - (total_complexity)

    # Clamp score between 0 and 100
    score = max(0, min(100, score))

    # Return the results:
    return {
