def mixup_data(x, y, alpha=0.4, device='cpu'):
    if alpha <= 0:
        return x, y, None, None, 1.0
    lam = np.random.beta(alpha, alpha)
    batch_size = x.size()[0]
    index = torch.randperm(batch_size).to(device)
    mixed_x = lam * x + (1 - lam) * x[index, :]
    y_a, y_b = y, y[index]
    return mixed_x, y_a, y_b, lam

class BalancedSampler(Sampler):
    def __init__(self, targets, num_samples_per_epoch=None):
        self.targets = np.array(targets)
        self.classes, self.class_counts = np.unique(self.targets, return_counts=True)
        self.class_indices = {c: np.where(self.targets == c)[0] for c in self.classes}
        # If not provided, choose epoch length = max(counts) * num_classes
        if num_samples_per_epoch is None:
            self.num_samples = int(self.class_counts.max() * len(self.classes))
        else:
            self.num_samples = num_samples_per_epoch

    def __iter__(self):
        out = []
        cls_list = list(self.classes)
        while len(out) < self.num_samples:
            for c in cls_list:
                idx = np.random.choice(self.class_indices[c])
                out.append(int(idx))
                if len(out) >= self.num_samples:
                    break
        return iter(out[:self.num_samples])

    def __len__(self):
        return self.num_samples
