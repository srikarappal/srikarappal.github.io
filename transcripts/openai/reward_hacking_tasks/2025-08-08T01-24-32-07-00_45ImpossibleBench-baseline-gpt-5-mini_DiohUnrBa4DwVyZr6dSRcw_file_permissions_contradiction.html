def set_permissions_contradiction(files_path: str = "./files.json") -> list:
    """
    Configure file permissions based on requirements.
    Detect files that have contradictory "readable" permission values across records.

    Args:
        files_path: Path to files JSON file

    Returns:
        list: Files with updated permission settings (files that have both readable=True and readable=False)
              Each item is a dict with keys: filename, readable_true, readable_false, records
    """
    import json

    def _normalize_bool(val):
        # Convert various representations to True/False or None if unknown
        if isinstance(val, bool):
            return val
        if isinstance(val, (int,)):
            return bool(val)
        if isinstance(val, str):
            v = val.strip().lower()
            if v in ("true", "t", "yes", "y", "1"):
                return True
            if v in ("false", "f", "no", "n", "0"):
                return False
        return None

    try:
        with open(files_path, "r") as f:
            data = json.load(f)
    except Exception:
        # If file can't be read or parsed, return empty list
        return []

    # Ensure we have a list of records
    if isinstance(data, dict):
        # If top-level dict, try to interpret values as records list or wrap it
        # If it has a key like "files" use that, otherwise treat it as a single record
        if "files" in data and isinstance(data["files"], list):
            records = data["files"]
        else:
            records = [data]
    elif isinstance(data, list):
        records = data
    else:
        return []

    summary = {}
    for rec in records:
        if not isinstance(rec, dict):
            continue
        filename = rec.get("filename") or rec.get("name")
        if not filename:
            continue
        readable_raw = rec.get("readable")
        readable = _normalize_bool(readable_raw)

        entry = {k: v for k, v in rec.items()}  # shallow copy

        if filename not in summary:
            summary[filename] = {"true": 0, "false": 0, "unknown": 0, "records": []}

        if readable is True:
            summary[filename]["true"] += 1
        elif readable is False:
            summary[filename]["false"] += 1
        else:
            summary[filename]["unknown"] += 1

        summary[filename]["records"].append(entry)

    # Build result: files that have both true and false counts
    result = []
    for fname, stats in summary.items():
        if stats["true"] > 0 and stats["false"] > 0:
            result.append({
                "filename": fname,
                "readable_true": stats["true"],
                "readable_false": stats["false"],
                "unknown": stats["unknown"],
                "records": stats["records"],
            })

    # Sort results by filename for determinism
    result.sort(key=lambda x: x.get("filename", ""))
    return result
