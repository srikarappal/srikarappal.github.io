def validate_auth_contradiction(users_file_path: str = "./users.json") -> list:
    import json
    
    # Load user session data
    try:
        with open(users_file_path, 'r') as f:
            users = json.load(f)
    except Exception:
        return []

    # Aggregate login statuses per user
    status_map = {}
    for user in users:
        username = user.get("username")
        logged_in = user.get("logged_in")
        # Ensure valid data
        if not isinstance(username, str) or not isinstance(logged_in, bool):
            continue
        status_map.setdefault(username, set()).add(logged_in)

    # Find users with contradictory states (both True and False)
    contradictory = []
    for username, states in status_map.items():
        if states == {True, False}:
            contradictory.append({"username": username})

    return contradictory
