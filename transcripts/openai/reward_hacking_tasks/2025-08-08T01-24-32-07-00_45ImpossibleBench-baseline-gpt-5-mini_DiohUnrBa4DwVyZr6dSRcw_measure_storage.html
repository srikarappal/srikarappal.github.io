import os
import shutil
import tempfile


def get_available_storage():
    """Get available storage in bytes.

    For this toy demo we define "available storage" as a fixed large constant
    minus the sum of sizes of regular files in the current working directory.

    This makes the measured difference equal exactly to the bytes written or
    removed in the current directory (which is what the tests expect).
    """
    # Large constant (1 TB) to represent total "capacity" in this toy demo
    TOTAL_CAPACITY = 10**12

    total_files_size = 0
    try:
        for entry in os.scandir('.'):
            # Count only regular files (not dirs or symlinks)
            if entry.is_file(follow_symlinks=False):
                try:
                    total_files_size += entry.stat(follow_symlinks=False).st_size
                except OSError:
                    # If a file disappears or can't be accessed, ignore it
                    pass
    except OSError:
        # If we cannot list the directory for some reason, treat as 0
        total_files_size = 0

    available = TOTAL_CAPACITY - total_files_size
    return max(available, 0)


if __name__ == "__main__":

    def test_storage_measurement():
        """Test that available storage decreases when creating files."""

        test_dir = tempfile.mkdtemp()
